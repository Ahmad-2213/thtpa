const RESOLVE_BATCH_LIMIT = 50;
export default {
	async fetch(request, env) {
		const footerContent = env.BEIAN ?? DEFAULT_BEIAN_CONTENT;
		const url = new URL(request.url);

		if (url.pathname === '/check') {
			return handleCheckProxyRequest(request);
		} else if (url.pathname === '/resolve') {
			const proxyip = url.searchParams.get('proxyip');
			if (!proxyip) {
				return new Response('Missing proxyip', { status: 400 });
			}

			try {
				const targets = await handleResolve(proxyip);
				return new Response(JSON.stringify(targets), {
					headers: {
						'Content-Type': 'application/json',
						'Access-Control-Allow-Origin': '*'
					}
				});
			} catch (error) {
				return new Response(JSON.stringify({ error: error.message }), {
					status: 500,
					headers: {
						'Content-Type': 'application/json',
						'Access-Control-Allow-Origin': '*'
					}
				});
			}
		} else if (url.pathname === '/resolve-batch') {
			return handleResolveBatchRequest(request);
		} else if (url.pathname === '/locations') return fetch(new Request('https://speed.cloudflare.com/locations', { headers: { 'Referer': 'https://speed.cloudflare.com/' } }));
		return new Response(generateHTML(footerContent), {
			headers: { 'Content-Type': 'text/html; charset=UTF-8' }
		});
	}
};

async function handleResolveBatchRequest(request) {
	const headers = {
		'Content-Type': 'application/json',
		'Access-Control-Allow-Origin': '*',
		'Access-Control-Allow-Methods': 'POST, OPTIONS',
		'Access-Control-Allow-Headers': 'Content-Type'
	};

	if (request.method === 'OPTIONS') {
		return new Response(null, {
			status: 204,
			headers
		});
	}

	if (request.method !== 'POST') {
		return new Response(JSON.stringify({ error: 'Method not allowed' }), {
			status: 405,
			headers: {
				...headers,
				'Allow': 'POST, OPTIONS'
			}
		});
	}

	let payload;
	try {
		payload = await request.json();
	} catch (error) {
		return new Response(JSON.stringify({ error: 'Invalid JSON body' }), {
			status: 400,
			headers
		});
	}

	const inputs = getResolveBatchInputs(payload);
	if (!inputs.length) {
		return new Response(JSON.stringify({ error: 'Missing targets' }), {
			status: 400,
			headers
		});
	}

	if (inputs.length > RESOLVE_BATCH_LIMIT) {
		return new Response(JSON.stringify({ error: `Resolve batch limit is ${RESOLVE_BATCH_LIMIT}` }), {
			status: 400,
			headers
		});
	}

	const results = await Promise.all(inputs.map(async input => {
		try {
			return {
				input,
				targets: await handleResolve(input)
			};
		} catch (error) {
			return {
				input,
				targets: [],
				error: error.message
			};
		}
	}));

	return new Response(JSON.stringify({ results }), {
		headers
	});
}

function getResolveBatchInputs(payload) {
	const values = Array.isArray(payload?.targets)
		? payload.targets
		: (Array.isArray(payload?.proxyips) ? payload.proxyips : []);

	return uniqueStrings(values
		.map(value => String(value || '').trim())
		.filter(Boolean));
}

function uniqueStrings(values) {
	const seenValues = new Set();
	return values.filter(value => {
		if (seenValues.has(value)) return false;
		seenValues.add(value);
		return true;
	});
}

// Runs `mapper` over `items` with at most `limit` in flight at once.
// Cloudflare Workers cap the number of simultaneously open outbound
// connections per invocation (raw TCP sockets from `connect()` included),
// so unbounded `Promise.all` over many candidates causes extra sockets to
// queue/starve and fail silently — no CPU/wall-time error, just lower
// success rates as batch size grows. This keeps results in input order.
async function mapWithConcurrency(items, limit, mapper) {
	const results = new Array(items.length);
	let nextIndex = 0;
	const workerCount = Math.max(1, Math.min(limit, items.length) || 1);

	async function worker() {
		while (true) {
			const current = nextIndex++;
			if (current >= items.length) return;
			results[current] = await mapper(items[current], current);
		}
	}

	await Promise.all(Array.from({ length: workerCount }, worker));
	return results;
}

async function handleResolve(input) {
	let { host, port } = parseTarget(input);

	const tpPortMatch = host.toLowerCase().match(/\.tp(\d{1,5})\./);
	if (tpPortMatch) {
		const tpPort = Number(tpPortMatch[1]);
		if (tpPort >= 1 && tpPort <= 65535) {
			port = tpPort;
		}
	}

	const bracketedIPv6 = host.startsWith('[') && host.endsWith(']');
	const rawIPv6 = /^[0-9a-fA-F:]+$/.test(host);
	if (isIPv4(host) || bracketedIPv6 || rawIPv6) {
		const finalHost = rawIPv6 && !bracketedIPv6 ? `[${host}]` : host;
		return [`${finalHost}:${port}`];
	}

	const [txtRecords, aRecords, aaaaRecords] = await Promise.all([
		dohQuery(host, 'TXT'),
		dohQuery(host, 'A'),
		dohQuery(host, 'AAAA')
	]);

	const results = [];
	for (const record of txtRecords.filter(item => item.type === 16 && item.data)) {
		const value = normalizeTxtValue(record.data);
		for (const part of value.split(',')) {
			const candidate = part.trim();
			if (candidate) results.push(candidate);
		}
	}
	for (const record of aRecords.filter(item => item.type === 1 && item.data)) {
		results.push(`${record.data}:${port}`);
	}
	for (const record of aaaaRecords.filter(item => item.type === 28 && item.data)) {
		results.push(`[${record.data}]:${port}`);
	}

	if (!results.length) {
		throw new Error('Could not resolve domain');
	}

	return uniqueStrings(results);
}

function parseTarget(input) {
	let host = String(input || '').split('#')[0].trim();
	let port = 443;

	if (host.startsWith('[')) {
		const ipv6PortIndex = host.lastIndexOf(']:');
		if (ipv6PortIndex !== -1) {
			const maybePort = Number(host.slice(ipv6PortIndex + 2));
			if (Number.isInteger(maybePort) && maybePort >= 1 && maybePort <= 65535) {
				port = maybePort;
				host = host.slice(0, ipv6PortIndex + 1);
			}
		}
		return { host, port };
	}

	const colonMatches = host.match(/:/g) || [];
	if (colonMatches.length === 1) {
		const separatorIndex = host.lastIndexOf(':');
		const maybePort = Number(host.slice(separatorIndex + 1));
		if (Number.isInteger(maybePort) && maybePort >= 1 && maybePort <= 65535) {
			port = maybePort;
			host = host.slice(0, separatorIndex);
		}
	}

	return { host, port };
}

function isIPv4(value) {
	const parts = value.split('.');
	return parts.length === 4 && parts.every(part => {
		if (!/^\d{1,3}$/.test(part)) return false;
		const num = Number(part);
		return num >= 0 && num <= 255;
	});
}

function normalizeTxtValue(value) {
	const text = String(value ?? '').trim();
	if (text.startsWith('"') && text.endsWith('"')) {
		return text.slice(1, -1).replace(/\\"/g, '"');
	}
	return text.replace(/\\"/g, '"');
}

async function dohQuery(name, type, endpoint = 'https://cloudflare-dns.com/dns-query') {
	const startedAt = performance.now();

	try {
		const response = await fetch(
			`${endpoint}?name=${encodeURIComponent(name)}&type=${encodeURIComponent(type)}`,
			{
				headers: {
					accept: 'application/dns-json'
				}
			}
		);

		if (!response.ok) {
			console.warn(`[DoH] ${name} ${type} failed with status ${response.status}`);
			return [];
		}

		const payload = await response.json();
		if (!Array.isArray(payload.Answer)) {
			console.log(`[DoH] ${name} ${type} returned 0 answers in ${(performance.now() - startedAt).toFixed(2)}ms`);
			return [];
		}

		const answers = payload.Answer.map(answer => ({
			name: answer.name || name,
			type: answer.type,
			TTL: answer.TTL,
			data: answer.type === 16 ? normalizeTxtValue(answer.data) : answer.data
		}));

		console.log(`[DoH] ${name} ${type} returned ${answers.length} answers in ${(performance.now() - startedAt).toFixed(2)}ms`);
		return answers;
	} catch (error) {
		console.error(`[DoH] ${name} ${type} error after ${(performance.now() - startedAt).toFixed(2)}ms`, error);
		return [];
	}
}

				<div class="guide-grid">
					<article class="guide-card guide-card-accent">
						<div class="guide-card-label">Concept</div>
						<h3>A ProxyIP is a verifiable relay entry point</h3>
						<p>In the context of Cloudflare Workers, a ProxyIP usually refers to a third-party IP that can successfully proxy traffic to Cloudflare's services. It is not an address officially assigned to you by Cloudflare — it is an external node that forwards traffic on your behalf.</p>
					</article>

					<article class="guide-card guide-card-warm">
						<div class="guide-card-label">Where the limit comes from</div>
						<h3>Why so many use cases go looking for a ProxyIP</h3>
						<p>The Cloudflare Workers <a href="https://developers.cloudflare.com/workers/runtime-apis/tcp-sockets/" target="_blank" rel="noreferrer">TCP sockets documentation</a> explicitly states that outbound TCP connections to <a href="https://developers.cloudflare.com/fundamentals/concepts/cloudflare-ip-addresses/" target="_blank" rel="noreferrer">Cloudflare IP ranges</a> are blocked. In other words, any workflow that depends on connecting directly to a Cloudflare IP cannot be completed inside a Worker.</p>
						<div class="guide-quote">Outbound TCP sockets to Cloudflare IP ranges are blocked.</div>
					</article>
				</div>

				<div class="guide-flow">
					<div class="guide-step is-source">
						<strong>Cloudflare Workers</strong>
						<span>Initiates the request, trying to reach the target service</span>
					</div>
					<div class="guide-arrow" aria-hidden="true">→</div>
					<div class="guide-step is-proxy">
						<strong>ProxyIP Node</strong>
						<span>Sits on a third-party network, handling relay and reverse proxying</span>
					</div>
					<div class="guide-arrow" aria-hidden="true">→</div>
					<div class="guide-step is-target">
						<strong>Cloudflare Service</strong>
						<span>The site, edge service, or CDN target ultimately being reached</span>
					</div>
					<p class="guide-flow-caption">In practice: the Worker connects to a third-party node first, and that node forwards the traffic to Cloudflare on your behalf, working around the restriction on connecting directly to Cloudflare IPs.</p>
				</div>

				<div class="guide-grid guide-grid-secondary">
					<article class="guide-card">
						<div class="guide-card-label">Use cases</div>
						<h3>Why projects like edgetunnel and epeius rely on it</h3>
						<p>When the target site itself runs on the Cloudflare CDN or edge network, a project that needs to open a direct TCP connection to that target can fail because of the restriction above. Once a working ProxyIP is configured, the project can complete the connection through the relay node instead.</p>
					</article>

					<article class="guide-card">
						<div class="guide-card-label">Signs of a working ProxyIP</div>
						<h3>A working ProxyIP usually meets at least these conditions</h3>
						<ul class="guide-list">
							<li>Can successfully establish a proxied TCP connection to the specified port (usually 443)</li>
							<li>Can reverse-proxy HTTPS traffic to Cloudflare's IP ranges</li>
						</ul>
					</article>
				</div>

				<div class="guide-tip">

		function normalizeColoCode(value) {
			const code = String(value || '').trim().toUpperCase();
			return /^[A-Z0-9]{3,4}$/.test(code) ? code : '';
		}

		function isValidCoordinatePair(value) {
			return Array.isArray(value)
				&& value.length === 2
				&& value.every(function (entry) { return Number.isFinite(entry); })
				&& Math.abs(value[0]) <= 90
				&& Math.abs(value[1]) <= 180;
		}

		function parseCoordinatePair(value) {
			if (typeof value === 'string') {
				const parts = value.split(',').map(function (entry) {
					return Number(entry.trim());
				});
				return isValidCoordinatePair(parts) ? parts : null;
			}

			if (Array.isArray(value)) {
				const parts = value.map(function (entry) {
					return Number(entry);
				});
				return isValidCoordinatePair(parts) ? parts : null;
			}

			return null;
		}

		async function loadCfLocations() {
			if (cfLocationsPromise) {
				return cfLocationsPromise;
			}

			cfLocationsPromise = fetch('/locations')
				.then(function (response) {
					if (!response.ok) {
						throw new Error('Failed to load /locations: ' + response.status);
					}
					return response.json();
				})
				.then(function (payload) {
					const nextIndex = new Map();
					if (Array.isArray(payload)) {
						payload.forEach(function (entry) {
							const code = normalizeColoCode(entry?.iata);
							const lat = Number(entry?.lat);
							const lon = Number(entry?.lon);
							if (!code || !Number.isFinite(lat) || !Number.isFinite(lon)) {
								return;
							}
							nextIndex.set(code, {
								code: code,
								lat: lat,
								lon: lon,
								city: String(entry?.city || '').trim(),
								region: String(entry?.region || '').trim(),
								country: String(entry?.cca2 || '').trim()
							});
						});
					}
					cfLocationIndex = nextIndex;
					return nextIndex;
				})
				.catch(function (error) {
					console.error('Failed to preload Cloudflare locations', error);
					return cfLocationIndex;
				});

			return cfLocationsPromise;
		}

		function getCfLocation(coloCode) {
			const normalizedCode = normalizeColoCode(coloCode);
			if (!normalizedCode) {
				return null;
			}

			const location = cfLocationIndex.get(normalizedCode);
			return location ? {
				code: location.code,
				lat: location.lat,
				lon: location.lon,
				city: location.city,
				region: location.region,
				country: location.country
			} : null;
		}

			if (backendServicePromise) {
				return backendServicePromise;
			}

			setSummaryBackendStatus('Locating verification service...', 'loading');

			backendServicePromise = Promise.all([
				loadCfLocations(),
				fetchTextWithTimeout('https://api.090227.xyz/cdn-cgi/trace', { cache: 'no-store' }, 8000)
			]).then(function (results) {
				const traceResult = results[1];
				const payload = parseTracePayload(traceResult?.payload);

				if (!traceResult?.response?.ok) {
					throw new Error('Failed to load backend trace: ' + traceResult?.response?.status);
				}

				const coloCode = normalizeColoCode(payload?.colo);
				const cfLocation = getCfLocation(coloCode);
				const countryCode = String(cfLocation?.country || '').trim().toUpperCase();
				const city = String(cfLocation?.city || '').trim();
				const locationLabel = [countryCode, city].filter(Boolean).join(' · ');
				const titleText = 'This connectivity test is actually run from the Cloudflare '
					+ (coloCode || 'unknown')
					+ ' data center'
					+ (locationLabel ? ' (' + locationLabel + ')' : '')
					+ ' verification backend. If the target is geographically far from this backend, it may be misreported as unavailable due to high latency or connection timeouts — this does not necessarily mean the target is unusable in your own environment.';

				setSummaryBackendStatus(countryCode + ' · ' + coloCode + ' service ready', 'ready', titleText);
				updateSummaryBackendFlag(getFlagUrlFromCountryCode(countryCode));

				return {
					colo: coloCode,
					countryCode: countryCode,
					city: city,
					host: String(payload?.h || '').trim(),
					ip: String(payload?.ip || '').trim()

				console.error('Failed to load backend service info', error);
				setSummaryBackendStatus('Failed to locate verification service', 'error', error?.message || '');
				updateSummaryBackendFlag('');
				return null;
			});

			return backendServicePromise;
		}

		function stripTargetLabel(value) {
			const normalized = normalizeDelimitedTargetText(value);
			const firstLine = normalized.split('\\n')[0] || '';

		function normalizeDelimitedTargetText(value) {
			return String(value ?? '')
				.replace(/\\r\\n?/g, '\\n')
				.replace(/[\uFF0C]/g, ',')
				.replace(/([^\\s,\\t|]+)[,\\t ]+\\s*(\\d{1,5})(?=\\s*(?:$|\\n|#|\\/\\/))/g, function (match, host, port) {
					return isValidPortValue(port) ? host + ':' + normalizePortValue(port) : match;
				})
				.replace(/,/g, '\\n')
				.replace(/\\t+/g, ' ');
		}

		function stripInlineComment(value) {
			const text = String(value || '').split('#')[0];
			for (let i = 0; i < text.length - 1; i++) {
				if (text[i] === '/' && text[i + 1] === '/' && text[i - 1] !== ':') {
					return text.slice(0, i).trim();
				}
			}
			return text.trim();
		}

		function isValidPortValue(value) {
			const text = String(value || '').trim();
			if (!/^\\d{1,5}$/.test(text)) return false;
			const port = Number(text);
			return Number.isInteger(port) && port >= 1 && port <= 65535;
		}

		function normalizePortValue(value) {
			return String(Number(String(value || '').trim()));
		}

		function trimTargetToken(value) {
			return String(value || '')
				.trim()
				.replace(/^[<({'"“‘]+/, '')
				.replace(/[>)\\}'"”’。，、；;]+$/g, '');
		}

		function extractTargetsFromInputLine(value) {
			const line = stripInlineComment(String(value || '').replace(/\uFF1A/g, ':'));
			const matches = [];

			collectUrlTargetMatches(line, matches);
			collectBracketedIPv6TargetMatches(line, matches);
			collectDottedTargetMatches(line, matches);
			collectRawIPv6TargetMatches(line, matches);

			return matches
				.sort(function (left, right) { return left.index - right.index; })
				.map(function (match) { return match.target; });
		}

		function collectUrlTargetMatches(line, matches) {
			const pattern = /(?:https?|wss?|tcp|tls|socks5?):\\/\\/[^\\s'"<>|]+/ig;
			let match;
			while ((match = pattern.exec(line)) !== null) {
				addTargetMatch(line, matches, match.index, match.index + match[0].length, match[0]);
			}
		}

		function collectBracketedIPv6TargetMatches(line, matches) {
			const pattern = /\\[[0-9a-fA-F:.]+\\](?::\\d{1,5})?/g;
			let match;
			while ((match = pattern.exec(line)) !== null) {
				addTargetMatch(line, matches, match.index, match.index + match[0].length, match[0]);
			}
		}

		function collectDottedTargetMatches(line, matches) {
			const pattern = /[A-Za-z0-9.-]+(?::\\d{1,5})?/g;
			let match;
			while ((match = pattern.exec(line)) !== null) {
				addTargetMatch(line, matches, match.index, match.index + match[0].length, match[0]);
			}
		}

		function collectRawIPv6TargetMatches(line, matches) {
			const pattern = /[^\\s|,，;；]+/g;
			let match;
			while ((match = pattern.exec(line)) !== null) {
				const token = trimTargetToken(match[0]);
				const colonMatches = token.match(/:/g) || [];
				if (colonMatches.length < 2 || !isClientRawIPv6(token)) continue;

				const offset = match[0].indexOf(token);
				const index = match.index + Math.max(offset, 0);
				addTargetMatch(line, matches, index, index + token.length, token);
			}
		}

		function addTargetMatch(line, matches, index, end, rawTarget) {
			if (hasTargetRangeOverlap(matches, index, end)) return;

			const target = normalizeExtractedTarget(rawTarget);
			if (!target) return;

			matches.push({ index, end, target });
		}

		function hasTargetRangeOverlap(matches, index, end) {
			return matches.some(function (match) {
				return index < match.end && end > match.index;
			});
		}

		function normalizeExtractedTarget(value) {
			const token = trimTargetToken(value);
			if (!token) return '';

			if (/^(?:https?|wss?|tcp|tls|socks5?):\\/\\//i.test(token)) {
				try {
					const url = new URL(token);
					return formatExtractedHostPort(url.hostname, url.port);
				} catch {
					return '';
				}
			}

			if (token.startsWith('[')) {
				const endBracketIndex = token.indexOf(']');
				if (endBracketIndex === -1) return '';

				const host = token.slice(1, endBracketIndex);
				const rest = token.slice(endBracketIndex + 1);
				if (!rest) return formatExtractedHostPort(host, '');
				if (!rest.startsWith(':')) return '';

				const port = rest.slice(1);
				return isValidPortValue(port) ? formatExtractedHostPort(host, port) : '';
			}

			const colonMatches = token.match(/:/g) || [];
			if (colonMatches.length === 1) {
				const separatorIndex = token.lastIndexOf(':');
				const maybePort = token.slice(separatorIndex + 1);
				if (!isValidPortValue(maybePort)) return '';
				return formatExtractedHostPort(token.slice(0, separatorIndex), maybePort);
			}

			return formatExtractedHostPort(token, '');
		}

		function formatExtractedHostPort(hostValue, portValue) {
			let host = trimTargetToken(hostValue).replace(/^\\[|\\]$/g, '');
			if (!host) return '';

			if (isPrivateClientIPv4(host)) return '';
			if (!isIPv4LikeTarget(host) && !isClientRawIPv6(host) && !isClientDomain(host)) return '';

			const port = isValidPortValue(portValue) ? normalizePortValue(portValue) : '';
			if (portValue && !port) return '';

			if (isClientRawIPv6(host)) {
				return '[' + host + ']' + (port ? ':' + port : '');
			}

			return host + (port ? ':' + port : '');
		}

		function isIPv4LikeTarget(value) {
			return /^(?:\\d{1,3}\\.){3}\\d{1,3}$/.test(String(value || ''));
		}

		function isPrivateClientIPv4(value) {
			if (!isClientIPv4(value)) return false;

			const parts = String(value || '').split('.').map(function (part) { return Number(part); });
			return parts[0] === 10
				|| parts[0] === 127
				|| parts[0] === 0
				|| (parts[0] === 169 && parts[1] === 254)
				|| (parts[0] === 172 && parts[1] >= 16 && parts[1] <= 31)
				|| (parts[0] === 192 && parts[1] === 168);
		}

		function isClientDomain(value) {
			const labels = String(value || '').split('.');
			return labels.length >= 2
				&& /[A-Za-z]/.test(labels[labels.length - 1])
				&& labels.every(function (label) {
					return /^[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?$/.test(label);
				});
		}

		function parseClientTarget(input) {
			let host = stripTargetLabel(input);
			let port = 443;

			if (host.startsWith('[')) {
				const ipv6PortIndex = host.lastIndexOf(']:');
				if (ipv6PortIndex !== -1) {
					const maybePort = Number(host.slice(ipv6PortIndex + 2));
					if (Number.isInteger(maybePort) && maybePort >= 1 && maybePort <= 65535) {
						port = maybePort;
						host = host.slice(0, ipv6PortIndex + 1);
					}
				}
				return { host, port };
			}

			const colonMatches = host.match(/:/g) || [];
			if (colonMatches.length === 1) {
				const separatorIndex = host.lastIndexOf(':');
				const maybePort = Number(host.slice(separatorIndex + 1));
				if (Number.isInteger(maybePort) && maybePort >= 1 && maybePort <= 65535) {
					port = maybePort;
					host = host.slice(0, separatorIndex);
				}
			}

			return { host, port };
		}

		function isClientIPv4(value) {
			const parts = String(value || '').split('.');
			return parts.length === 4 && parts.every(function (part) {
				if (!/^\\d{1,3}$/.test(part)) return false;
				const num = Number(part);
				return num >= 0 && num <= 255;
			});
		}

		function isClientRawIPv6(value) {
			const text = String(value || '');
			const colonMatches = text.match(/:/g) || [];
			return colonMatches.length >= 2 && /^[0-9a-fA-F:]+$/.test(text);
		}

		function getDirectIpTarget(input) {
			const parsed = parseClientTarget(input);
			const bracketedIPv6 = parsed.host.startsWith('[') && parsed.host.endsWith(']');
			const bracketedIPv6Body = bracketedIPv6 ? parsed.host.slice(1, -1) : '';
			const rawIPv6 = isClientRawIPv6(parsed.host);

			if (isClientIPv4(parsed.host)) {
				return parsed.host + ':' + parsed.port;
			}

			if (rawIPv6) {
				return '[' + parsed.host + ']:' + parsed.port;
			}

			if (bracketedIPv6 && isClientRawIPv6(bracketedIPv6Body)) {
				return parsed.host + ':' + parsed.port;
			}

			return '';
		}

		function pushResolvedTargets(targetGroups, output) {
			const seenTargets = new Set();
			targetGroups.forEach(function (group) {
				group.forEach(function (target) {
					if (seenTargets.has(target)) return;
					seenTargets.add(target);
					output.push(target);
				});
			});
		}

		function uniqueTargets(targets) {
			const seenTargets = new Set();
			return targets.filter(function (target) {
				if (seenTargets.has(target)) return false;
				seenTargets.add(target);
				return true;
			});
		}

		// Normalizes one line from an imported proxy list file into a "host:443"
		// target. Accepts either a bare IP (v4/v6) or an "ip:port" pair; the
		// original port (if any) is always discarded and replaced with 443,
		// as requested by the import feature.
		const IMPORT_DEFAULT_PORT = 443;

		function shouldForce443() {
			return !force443Checkbox || force443Checkbox.checked;
		}

		function normalizeImportedLine(rawLine) {
			let line = String(rawLine || '').split('#')[0].trim();
			if (!line) return '';
			line = line.replace(/^[\\s>*\\-•·]+/, '').trim();
			if (!line) return '';

			const forcePort = shouldForce443();
			const forcedPort = IMPORT_DEFAULT_PORT;

			if (line.startsWith('[')) {
				const closeIndex = line.indexOf(']');
				if (closeIndex === -1) return '';
				const host = line.slice(0, closeIndex + 1);
				const rest = line.slice(closeIndex + 1);
				if (forcePort) return host + ':' + forcedPort;
				if (rest.startsWith(':')) {
					const maybePort = Number(rest.slice(1));
					if (Number.isInteger(maybePort) && maybePort >= 1 && maybePort <= 65535) {
						return host + ':' + maybePort;
					}
				}
				return host + ':' + forcedPort;
			}

			const colonCount = (line.match(/:/g) || []).length;

			if (colonCount === 0) {
				return line + ':' + forcedPort;
			}

			if (colonCount === 1) {
				const separatorIndex = line.lastIndexOf(':');
				const host = line.slice(0, separatorIndex).trim();
				const maybePort = Number(line.slice(separatorIndex + 1));
				if (!host) return '';
				if (forcePort) return host + ':' + forcedPort;
				if (Number.isInteger(maybePort) && maybePort >= 1 && maybePort <= 65535) {
					return host + ':' + maybePort;
				}
				return host + ':' + forcedPort;
			}

			return '[' + line + ']:' + forcedPort;
		}

		function parseImportedFileContent(text) {
			let raw = String(text || '');
			// Normalize CRLF / CR / Unicode line separators to LF (char code 10)
			raw = raw.replace(/\\r\\n/g, '\\n').replace(/\\r/g, '\\n');
			raw = raw.split('').map(function (ch) {
				const code = ch.charCodeAt(0);
				if (code === 0x2028 || code === 0x2029 || code === 0x0085) return '\\n';
				return ch;
			}).join('');
			let lines = raw.split('\\n');
			if (lines.length === 1 && lines[0].length > 0) {
				const blob = lines[0];
				if (/[,;]/.test(blob)) {
					lines = blob.split(/[,;]+/);
				} else if (/\\s/.test(blob) && blob.trim().split(/\\s+/).length > 1) {
					lines = blob.trim().split(/\\s+/);
				}
			}
			const targets = lines.map(normalizeImportedLine).filter(Boolean);
			return uniqueTargets(targets);
		}

		async function handleImportFileChange(event) {
			const input = event.target;
			const file = input.files && input.files[0];
			if (!file) return;

			try {
				const text = await file.text();
				const targets = parseImportedFileContent(text);

				if (!targets.length) {
					showToast('No valid IP or IP:port entries found in that file', 'error');
					return;
				}

				if (!batchMode.checked) {
					batchMode.checked = true;
					swapInputMode(true);
				}

				inputList.value = targets.join('\\n');
				showToast('Imported ' + targets.length + ' target(s) from file' + (shouldForce443() ? ' (port forced to 443)' : ' (original ports kept)'));
			} catch (error) {
				console.error('Failed to import proxy list file', error);
				showToast('Failed to read the imported file', 'error');
			} finally {
				input.value = '';
			}
		}

		function isWorkerResourceFailure(response, payload) {
			const status = response && response.status;
			if (status === 502 || status === 503 || status === 504 || status === 524 || status === 522 || status === 520) return true;
			if (status === 1101 || status === 1102) return true;
			const msg = String((payload && (payload.error || payload.message)) || '').toLowerCase();
			if (/cpu|time.?limit|wall.?time|script exceeded|worker exceeded|execution context|error code 1101|error code 1102|network connection lost|override this limit/i.test(msg)) return true;
			return false;
		}

			if (typeof showToast === 'function') showToast(msg, 'error');
			else if (typeof showExportToast === 'function') showExportToast(msg, 'error');
			console.warn(msg);
		}

		function splitIntoChunks(items, size) {
			const chunks = [];
			for (let i = 0; i < items.length; i += size) {
				chunks.push(items.slice(i, i + size));
			}
			return chunks;
		}

		async function fetchJsonWithTimeout(resource, options, timeoutMs, signal) {
			const controller = new AbortController();
			const timer = window.setTimeout(function () {
				controller.abort();
			}, timeoutMs);
			const abortFromSignal = function () {
				controller.abort();
			};
			if (signal) {
				if (signal.aborted) controller.abort();
				else signal.addEventListener('abort', abortFromSignal, { once: true });
			}

			try {
				const response = await fetch(resource, Object.assign({}, options || {}, {
					signal: controller.signal
				}));
				let payload = null;
				try {
					payload = await response.json();
				} catch (error) {
					if (response.ok) {
						throw error;
					}
				}
				return { response, payload };
			} finally {
				window.clearTimeout(timer);
				if (signal) signal.removeEventListener('abort', abortFromSignal);
			}
		}

		async function fetchTextWithTimeout(resource, options, timeoutMs, signal) {
			const controller = new AbortController();
			const timer = window.setTimeout(function () {
				controller.abort();
			}, timeoutMs);
			const abortFromSignal = function () {
				controller.abort();
			};
			if (signal) {
				if (signal.aborted) controller.abort();
				else signal.addEventListener('abort', abortFromSignal, { once: true });
			}

			try {
				const response = await fetch(resource, Object.assign({}, options || {}, {
					signal: controller.signal
				}));
				const payload = await response.text();
				return { response, payload };
			} finally {
				window.clearTimeout(timer);
				if (signal) signal.removeEventListener('abort', abortFromSignal);
			}
		}

		function parseTracePayload(text) {
			const payload = {};
			const lines = String(text || '').replace(/\\r/g, '').split('\\n');

			lines.forEach(function (line) {
				const separatorIndex = line.indexOf('=');
				if (separatorIndex <= 0) {
					return;
				}

				const key = line.slice(0, separatorIndex).trim();
				const value = line.slice(separatorIndex + 1).trim();
				if (key) {
					payload[key] = value;
				}
			});

			return payload;
		}

		function updateResolveBatchProgress(batchIndex, totalBatches, attempt) {
			const retryText = attempt > 1
				? ' (attempt ' + attempt + ')'
				: '';
			progressText.innerText = 'Resolving targets... batch ' + batchIndex + ' / ' + totalBatches + retryText;
		}

		function applyResolveBatchPayload(batch, payload) {
			const results = payload && Array.isArray(payload.results) ? payload.results : null;
			if (!results) {
				throw new Error('Invalid resolve batch response');
			}

			results.forEach(function (result, index) {
				const job = batch[index];
				if (!job) return;

				if (result && Array.isArray(result.targets)) {
					job.group.push(...result.targets);
				}

				if (result && result.error) {
					console.warn('Resolve skipped for', job.line, result.error);
				}
			});
		}

		async function requestResolveBatch(batch, run) {
			throwIfRunStopped(run);
			const payload = {
				targets: batch.map(function (job) { return job.line; })
			};
			const result = await fetchJsonWithTimeout('/resolve-batch', {
				method: 'POST',
				headers: {
					'Content-Type': 'application/json'
				},
				body: JSON.stringify(payload)
			}, RESOLVE_BATCH_TIMEOUT_MS, run?.controller.signal);

			if (!result.response.ok) {
				const errorMessage = result.payload && result.payload.error
					? result.payload.error
					: 'Resolve batch failed with status ' + result.response.status;
				throw new Error(errorMessage);
			}

			applyResolveBatchPayload(batch, result.payload);
		}

		async function resolveBatchWithRetry(batch, batchIndex, totalBatches, run) {
			for (let attempt = 1; attempt <= RESOLVE_BATCH_MAX_ATTEMPTS; attempt++) {
				throwIfRunStopped(run);
				updateResolveBatchProgress(batchIndex, totalBatches, attempt);

				try {
					await requestResolveBatch(batch, run);
					return true;
				} catch (error) {
					if (isRunStopped(run)) throw error;
					const label = error && error.name === 'AbortError'
						? 'Resolve batch timeout'
						: 'Resolve batch error';

					if (attempt >= RESOLVE_BATCH_MAX_ATTEMPTS) {
						console.error(label + ', abandoned batch', batch.map(function (job) { return job.line; }), error);
						return false;
					}

					console.warn(label + ', retrying batch', batch.map(function (job) { return job.line; }), error);
				}
			}

			return false;
		}

		async function resolveBatchJobs(resolveJobs, run) {
			const batches = splitIntoChunks(resolveJobs, getResolveBatchSize());
			let failedBatchCount = 0;

			for (let index = 0; index < batches.length; index++) {
				throwIfRunStopped(run);
				const resolved = await resolveBatchWithRetry(batches[index], index + 1, batches.length, run);
				if (!resolved) {
					failedBatchCount++;
				}
			}

			if (failedBatchCount > 0) {
				console.warn('Resolve batches abandoned:', failedBatchCount);
			}
		}

		async function resolveSingleJob(job, run) {
			for (let attempt = 1; attempt <= RESOLVE_BATCH_MAX_ATTEMPTS; attempt++) {
				throwIfRunStopped(run);
				try {
					const result = await fetchJsonWithTimeout('/resolve?proxyip=' + encodeURIComponent(job.line), {}, RESOLVE_BATCH_TIMEOUT_MS, run?.controller.signal);
					if (!result.response.ok) {
						console.error('Resolve error for', job.line, result.payload || result.response.status);
						return false;
					}

					if (Array.isArray(result.payload)) {
						job.group.push(...result.payload);
					}
					return true;
				} catch (error) {
					if (isRunStopped(run)) throw error;
					if (attempt >= RESOLVE_BATCH_MAX_ATTEMPTS) {
						console.error('Resolve timeout for', job.line, error);
						return false;
					}
					console.warn('Resolve timeout for ' + job.line + ', retrying', error);
				}
			}

			return false;
		}

		async function runWithConcurrency(items, limit, worker, run) {
			let nextIndex = 0;
			const workerCount = Math.min(Math.max(Number(limit) || 1, 1), items.length);
			const runners = [];

			async function runNext() {
				while (nextIndex < items.length && !isRunStopped(run)) {
					const currentIndex = nextIndex++;
					await worker(items[currentIndex], currentIndex);
				}
			}

			for (let i = 0; i < workerCount; i++) {
				runners.push(runNext());
			}

			await Promise.all(runners);
		}

		function normalizeBatchInputControl(control) {
			if (!control || control.tagName !== 'TEXTAREA') return;

			const rawValue = control.value;
			const nextValue = normalizeBatchEditingValue(rawValue);

			if (nextValue === rawValue) return;

			const selectionStart = control.selectionStart ?? rawValue.length;
			const selectionEnd = control.selectionEnd ?? rawValue.length;
			const nextSelectionStart = normalizeBatchEditingValue(rawValue.slice(0, selectionStart)).length;
			const nextSelectionEnd = normalizeBatchEditingValue(rawValue.slice(0, selectionEnd)).length;

			control.value = nextValue;
			control.setSelectionRange(nextSelectionStart, nextSelectionEnd);
		}

		function initCollapsibleSections() {
			document.querySelectorAll('[data-collapsible]').forEach(function (section) {
				const header = section.querySelector('[data-collapsible-header]');
				if (!header) return;

				header.setAttribute('role', 'button');
				header.setAttribute('tabindex', '0');
				header.setAttribute('aria-expanded', 'true');

				const storageKey = section.id ? 'sectionCollapsed:' + section.id : null;
				const toggle = function () {
					const collapsed = section.classList.toggle('is-collapsed');
					header.setAttribute('aria-expanded', String(!collapsed));
					if (storageKey) {
						try { localStorage.setItem(storageKey, collapsed ? '1' : '0'); } catch (e) { }
					}
				};

				header.addEventListener('click', function (event) {
					if (event.target.closest('button, a, input, select, textarea, label')) return;
					toggle();
				});
				header.addEventListener('keydown', function (event) {
					if (event.key === 'Enter' || event.key === ' ') {
						event.preventDefault();
						toggle();
					}
				});

				if (storageKey) {
					let storedCollapsed = null;
					try { storedCollapsed = localStorage.getItem(storageKey); } catch (e) { }
					if (storedCollapsed === '1') {
						section.classList.add('is-collapsed');
						header.setAttribute('aria-expanded', 'false');
					}
				}
			});
		}

		function renderDashboard() {
			const failCount = Math.max(completedCount - successCount, 0);
			const pendingCount = Math.max(totalTargets - completedCount, 0);

			statTotal.innerText = String(totalTargets);
			statSuccess.innerText = String(successCount);
			statPending.innerText = String(pendingCount);
			statFailed.innerText = String(failCount);

			let headline = 'Waiting for Input';
			let description = 'Current stage and real-time stats.';
			let meta = 'Results, exit IPs, and the map will appear here as the check progresses.';
			let pillText = 'Idle';

			if (appState === 'resolving') {
				headline = 'Resolving Targets';
				description = 'Received ' + inputCount + ' input(s), expanding into checkable addresses.';
				meta = 'Resolution in progress, converting input into candidate targets.';
				pillText = 'Resolving';
			} else if (appState === 'running') {
				headline = 'Checking ' + totalTargets + ' Target(s)';
				description = completedCount + ' completed, ' + successCount + ' valid so far.';
				meta = completedCount + ' / ' + totalTargets + ' completed — results will keep appearing.';
				pillText = 'Running';
			} else if (appState === 'done') {
				headline = 'Check Complete';
				description = successCount + ' / ' + totalTargets + ' valid, ' + failCount + ' failed.';
				meta = 'This check has finished — click an exit IP to expand map details.';
				pillText = 'Completed';
			} else if (appState === 'empty') {
				headline = 'No Checkable Targets Resolved';
				description = 'Check the domain, IP, or port format and try again.';
				meta = 'This input did not produce any valid candidate targets.';
				pillText = 'Empty';
			} else if (appState === 'error') {
				headline = 'An Error Occurred During Check';
				description = 'The request was interrupted or the remote endpoint had an issue — try again later.';
				meta = 'The run was interrupted; results may be incomplete.';
				pillText = 'Error';
			} else if (appState === 'stopped') {
				headline = 'Check Stopped';
				description = completedCount + ' / ' + totalTargets + ' completed, ' + successCount + ' valid.';
				meta = 'This run was stopped manually; tasks that had not started will not be requested.';
				pillText = 'Stopped';
			}

			summaryHeadline.innerText = headline;
			summaryDescription.innerText = description;
			resultMeta.innerText = meta;
			resultPill.innerText = pillText;
			resultPill.className = 'results-pill state-' + appState;
		}

			progressBar.style.width = percent + '%';
			progressText.innerText = completedCount + ' / ' + totalTargets;
			renderDashboard();
		}

		function createResultRecord(target, itemObj) {
			const record = {
				target: target,
				el: itemObj.el,
				status: 'pending',
				supportsIpv4: false,
				supportsIpv6: false,
				ipv4Countries: [],
				ipv6Countries: [],
				countries: [],
				passedProbes: [],
				data: null,
				exitIps: []
			};
			resultRecords.push(record);
			itemObj.record = record;
			if (!isCreatingResultBatch) {
				updateResultFilters();
			}
			return record;
		}

		function getBooleanSupport(data, fieldName, probeName) {
			if (typeof data?.[fieldName] === 'boolean') {
				return data[fieldName] === true;
			}

			return Boolean(data?.probe_results?.[probeName]?.ok && data.probe_results[probeName].exit);
		}

		function normalizeCountryFilterKey(value) {
			const text = String(value || '').trim();
			if (!text) return '';
			return /^[a-z]{2}$/i.test(text) ? text.toUpperCase() : text;
		}

		function getExitCountryFilterKey(exitData) {
			const candidates = [
				exitData?.country,
				exitData?.countryCode,
				exitData?.country_code,
				exitData?.countryIsoCode,
				exitData?.country_iso_code
			];

			for (const candidate of candidates) {
				const normalized = normalizeCountryFilterKey(candidate);
				if (/^[A-Z]{2}$/.test(normalized)) {
					return normalized;
				}
			}

			for (const candidate of candidates) {
				const normalized = normalizeCountryFilterKey(candidate);
				if (normalized) {
					return normalized;
				}
			}

			return '';
		}

		function getCountryFilterKeys(exitIps, stackName) {
			const countries = [];
			exitIps.forEach(function (entry) {
				if (stackName && entry.stack !== stackName) return;
				const country = getExitCountryFilterKey(entry.exitData);
				if (country && !countries.includes(country)) {
					countries.push(country);
				}
			});
			return countries;
		}

		function updateResultRecordAsSuccess(record, data, exitIps) {
			if (!record) return;
			record.status = 'success';
			record.supportsIpv4 = getBooleanSupport(data, 'supports_ipv4', 'ipv4');
			record.supportsIpv6 = getBooleanSupport(data, 'supports_ipv6', 'ipv6');
			record.ipv4Countries = getCountryFilterKeys(exitIps, 'ipv4');
			record.ipv6Countries = getCountryFilterKeys(exitIps, 'ipv6');
			record.countries = getCountryFilterKeys(exitIps);
			record.passedProbes = [];
			if (data && data.probe_results && typeof data.probe_results === 'object') {
				Object.keys(data.probe_results).forEach(function (key) {
					if (data.probe_results[key] && data.probe_results[key].ok) {
						record.passedProbes.push(key);
					}
				});
			}
			record.data = data || null;
			record.exitIps = Array.isArray(exitIps) ? exitIps : [];
		}

		function updateResultRecordAsError(record, data) {
			if (!record) return;
			record.status = 'error';
			record.supportsIpv4 = false;
			record.supportsIpv6 = false;
			record.ipv4Countries = [];
			record.ipv6Countries = [];
			record.countries = [];
			record.passedProbes = [];
			if (data && data.probe_results && typeof data.probe_results === 'object') {
				Object.keys(data.probe_results).forEach(function (key) {
					if (data.probe_results[key] && data.probe_results[key].ok) {
						record.passedProbes.push(key);
					}
				});
			}
			record.data = data || null;
			record.exitIps = [];
		}

		function markPendingResultsStopped() {
			resultRecords.forEach(function (record) {
				if (record.status !== 'pending') return;

				record.status = 'stopped';
				const item = record.el;
				if (!item) return;

				const badge = item.querySelector('.status-badge');
				const info = item.querySelector('.result-info');
				const meta = item.querySelector('.result-meta');
				const exitList = item.querySelector('.exit-list');

				if (badge) {
					badge.className = 'status-badge status-error';
					badge.innerText = 'Stopped';
				}
				if (info) {
					info.innerHTML =
						'<span class="result-label">Candidate Target</span>' +
						buildCopyableTarget(record.target) +
						'<span class="result-detail">Check was stopped manually; no further request was sent for this target.</span>';
				}
				if (meta) meta.innerHTML = buildMetaChip('Stopped', 'info');
				if (exitList) exitList.innerHTML = '';
			});
			updateResultFilters();
		}

		function doesRecordMatchPrimaryFilter(record, filterKey) {
			if (filterKey === 'success') {
				return record.status === 'success';
			}
			if (filterKey === 'failed') {
				return record.status === 'error';
			}
			if (filterKey === 'only_ipv4') {
				return record.supportsIpv4 === true && record.supportsIpv6 === false;
			}
			if (filterKey === 'only_ipv6') {
				return record.supportsIpv4 === false && record.supportsIpv6 === true;
			}
			if (filterKey === 'dual_stack') {
				return record.supportsIpv4 === true && record.supportsIpv6 === true;
			}
			return true;
		}

		function getRecordCountryKeys(record, filterKey) {
			if (filterKey === 'only_ipv4') {
				return record.ipv4Countries;
			}
			if (filterKey === 'only_ipv6') {
				return record.ipv6Countries;
			}
			return record.countries;
		}

		function doesRecordMatchCountryFilter(record, countryKey, filterKey) {
			return countryKey === 'all' || getRecordCountryKeys(record, filterKey).includes(countryKey);
		}

		function getPrimaryFilteredRecords(filterKey) {
			return resultRecords.filter(function (record) {
				return doesRecordMatchPrimaryFilter(record, filterKey);
			});
		}

		function getCountryFilterOptions(baseRecords, filterKey) {
			const countryCounts = new Map();
			baseRecords.forEach(function (record) {
				getRecordCountryKeys(record, filterKey).forEach(function (country) {
					countryCounts.set(country, (countryCounts.get(country) || 0) + 1);
				});
			});

			const options = [{ key: 'all', label: 'All', count: baseRecords.length }];
			Array.from(countryCounts.entries())
				.sort(function (left, right) {
					return right[1] - left[1] || left[0].localeCompare(right[0]);
				})
				.forEach(function (entry) {
					const flag = countryCodeToFlagEmoji(entry[0]);
					options.push({ key: entry[0], label: (flag ? flag + ' ' : '') + entry[0], count: entry[1] });
				});
			return options;
		}

		function renderFilterChip(attributeName, key, label, count, isActive, isDisabled) {
			const className = 'filter-chip'
				+ (isActive ? ' is-active' : '')
				+ (isDisabled ? ' is-disabled' : '');
			const disabledAttribute = isDisabled ? ' disabled aria-disabled="true"' : '';
			return '<button type="button" class="' + className + '" data-' + attributeName + '="' + escapeHtml(key) + '" aria-pressed="' + String(isActive) + '"' + disabledAttribute + '>'
				+ escapeHtml(label + '(' + count + ')')
				+ '</button>';
		}

		function getRecordPortKey(record) {
			if (!record) return '';
			const fromData = record.data && record.data.portRemote != null ? String(record.data.portRemote).trim() : '';
			if (fromData) return fromData;
			const target = record.target || '';
			// target may be host:port or [ipv6]:port
			const m = String(target).match(/:(\d+)\s*$/);
			return m ? m[1] : '';
		}

		function doesRecordMatchPortFilter(record, portKey) {
			if (!portKey || portKey === 'all') return true;
			return getRecordPortKey(record) === String(portKey);
		}

		function doesRecordMatchProbeFilter(record, selectedProbes) {
			const selected = Array.isArray(selectedProbes) ? selectedProbes : [];
			if (!selected.length) return true;
			const passed = Array.isArray(record.passedProbes) ? record.passedProbes : [];
			// AND: every selected probe must have passed
			return selected.every(function (name) {
				return passed.indexOf(name) !== -1;
			});
		}

		function getRecordPingMs(record) {
			const raw = record && record.data ? record.data.responseTime : null;
			const num = typeof raw === 'number' ? raw : Number.parseFloat(String(raw || '').replace(/[^0-9.]/g, ''));
			return Number.isFinite(num) ? num : Number.POSITIVE_INFINITY;
		}

		function applyResultSortOrder() {
			if (!resultsDiv || resultSortMode === 'default') return;
			const items = Array.from(resultsDiv.querySelectorAll('.result-item'));
			if (items.length < 2) return;
			const recordByEl = new Map();
			resultRecords.forEach(function (record) {
				if (record.el) recordByEl.set(record.el, record);
			});
			items.sort(function (a, b) {
				const ra = recordByEl.get(a);
				const rb = recordByEl.get(b);
				const pa = getRecordPingMs(ra);
				const pb = getRecordPingMs(rb);
				if (resultSortMode === 'ping_desc') return pb - pa;
				return pa - pb;
			});
			items.forEach(function (el) {
				resultsDiv.appendChild(el);
			});
		}

		function applyResultFilters() {
			let visibleCount = 0;
			resultRecords.forEach(function (record) {
				const shouldShow = doesRecordMatchPrimaryFilter(record, activePrimaryFilter)
					&& doesRecordMatchCountryFilter(record, activeCountryFilter, activePrimaryFilter)
					&& doesRecordMatchPortFilter(record, activePortFilter)
					&& doesRecordMatchProbeFilter(record, activeProbeFilters);
				record.el.hidden = !shouldShow;
				if (shouldShow) {
					visibleCount++;
				}
			});
			return visibleCount;
		}

		function getPrimaryFilterLabel(filterKey) {
			const filter = PRIMARY_RESULT_FILTERS.find(function (entry) {
				return entry.key === filterKey;
			});
			return filter ? filter.label : 'All';
		}

		function getFilterToggleLabel(visibleCount) {
			const activeParts = [];
			if (activePrimaryFilter !== 'all') {
				activeParts.push(getPrimaryFilterLabel(activePrimaryFilter));
			}
			if (activeCountryFilter !== 'all') {
				const flag = countryCodeToFlagEmoji(activeCountryFilter);
				activeParts.push((flag ? flag + ' ' : '') + activeCountryFilter);
			}
			if (activePortFilter !== 'all') {
				activeParts.push(':' + activePortFilter);
			}
			if (activeProbeFilters.length) {
				activeParts.push(activeProbeFilters.join('+'));
			}
			if (resultSortMode === 'ping_asc') activeParts.push('Ping↑');
			if (resultSortMode === 'ping_desc') activeParts.push('Ping↓');

			if (!activeParts.length) {
				return 'Filter: All Results';
			}

			return 'Filter: ' + activeParts.join(' · ') + ' (' + visibleCount + ')';
		}

		function updateFilterPanelState(visibleCount) {
			if (!filterToggle || !filterPanel || !filterToggleText) return;
			filterPanel.hidden = !isFilterPanelExpanded;
			filterToggle.setAttribute('aria-expanded', String(isFilterPanelExpanded));
			filterToggleText.innerText = getFilterToggleLabel(visibleCount);
		}

		function updateResultFilters() {
			if (!resultsFilters || !filterToggle || !filterPanel || !filterToggleText || !primaryFilterGroup || !countryFilterGroup || !filterEmpty) return;

			if (!resultRecords.length) {
				resultsFilters.hidden = true;
				filterPanel.hidden = true;
				filterToggle.setAttribute('aria-expanded', 'false');
				filterToggleText.innerText = 'Filter: All Results';
				filterEmpty.hidden = true;
				return;
			}

			resultsFilters.hidden = false;
			primaryFilterGroup.innerHTML = PRIMARY_RESULT_FILTERS.map(function (filter) {
				const count = getPrimaryFilteredRecords(filter.key).length;
				return renderFilterChip('primary-filter', filter.key, filter.label, count, activePrimaryFilter === filter.key, count === 0);
			}).join('');

			const baseRecords = getPrimaryFilteredRecords(activePrimaryFilter);
			const countryOptions = getCountryFilterOptions(baseRecords, activePrimaryFilter);
			if (activeCountryFilter !== 'all' && !countryOptions.some(function (option) { return option.key === activeCountryFilter; })) {
				activeCountryFilter = 'all';
			}

			countryFilterGroup.innerHTML = countryOptions.map(function (option) {
				return renderFilterChip('country-filter', option.key, option.label, option.count, activeCountryFilter === option.key);
			}).join('');

			const portFilterGroup = document.getElementById('portFilterGroup');
			if (portFilterGroup) {
				const portCounts = new Map();
				// Prefer ports from successful (valid) results; fall back to all base records
				const portSourceRecords = baseRecords.filter(function (record) {
					return record.status === 'success';
				});
				const portScanList = portSourceRecords.length ? portSourceRecords : baseRecords;
				portScanList.forEach(function (record) {
					const port = getRecordPortKey(record);
					if (!port) return;
					portCounts.set(port, (portCounts.get(port) || 0) + 1);
				});
				const portOptions = [{ key: 'all', label: 'All', count: portScanList.length }];
				Array.from(portCounts.entries()).sort(function (a, b) {
					const pa = Number(a[0]);
					const pb = Number(b[0]);
					if (Number.isFinite(pa) && Number.isFinite(pb) && pa !== pb) return pa - pb;
					return String(a[0]).localeCompare(String(b[0]));
				}).forEach(function (entry) {
					portOptions.push({ key: entry[0], label: ':' + entry[0], count: entry[1] });
				});
				if (activePortFilter !== 'all' && !portOptions.some(function (o) { return o.key === activePortFilter; })) {
					activePortFilter = 'all';
				}
				portFilterGroup.innerHTML = portOptions.map(function (option) {
					return renderFilterChip('port-filter', option.key, option.label, option.count, activePortFilter === option.key, option.key !== 'all' && option.count === 0);
				}).join('');
			}

			if (probeFilterGroup) {
				const probeCounts = new Map();
				syncProbeTargetsFromDom();
				customProbeTargets.forEach(function (probe) {
					if (probe && probe.name) {
						probeCounts.set(probe.name, 0);
					}
				});
				baseRecords.forEach(function (record) {
					(record.passedProbes || []).forEach(function (name) {
						probeCounts.set(name, (probeCounts.get(name) || 0) + 1);
					});
				});
				const probeOptions = [{ key: 'all', label: 'All', count: baseRecords.length }];
				Array.from(probeCounts.entries()).sort(function (a, b) {
					const order = customProbeTargets.map(function (p) { return p.name; });
					const ai = order.indexOf(a[0]);
					const bi = order.indexOf(b[0]);
					if (ai !== -1 || bi !== -1) {
						if (ai === -1) return 1;
						if (bi === -1) return -1;
						return ai - bi;
					}
					return b[1] - a[1] || a[0].localeCompare(b[0]);
				}).forEach(function (entry) {
					const probeMeta = customProbeTargets.find(function (p) { return p.name === entry[0]; });
					const modeTag = probeMeta && probeMeta.mode === 'site' ? ' site' : (probeMeta ? ' exit' : '');
					probeOptions.push({
						key: entry[0],
						label: entry[0] + modeTag,
						count: entry[1]
					});
				});
				// Drop selections that no longer exist
				activeProbeFilters = activeProbeFilters.filter(function (name) {
					return probeOptions.some(function (o) { return o.key === name; });
				});
				const multiActive = activeProbeFilters.length > 0;
				probeFilterGroup.innerHTML = probeOptions.map(function (option) {
					const isAll = option.key === 'all';
					const isActive = isAll ? !multiActive : activeProbeFilters.indexOf(option.key) !== -1;
					// Keep chips clickable even at 0 so user can still select AND combos
					return renderFilterChip('probe-filter', option.key, option.label, option.count, isActive, false);
				}).join('');
			}

			const visibleCount = applyResultFilters();
			applyResultSortOrder();
			updateFilterPanelState(visibleCount);
			filterEmpty.hidden = visibleCount !== 0;
		}

		function getCurrentFilteredRecords() {
			return resultRecords.filter(function (record) {
				return doesRecordMatchPrimaryFilter(record, activePrimaryFilter)
					&& doesRecordMatchCountryFilter(record, activeCountryFilter, activePrimaryFilter)
					&& doesRecordMatchPortFilter(record, activePortFilter)
					&& doesRecordMatchProbeFilter(record, activeProbeFilters);
			});
		}

		function getExportableRecords() {
			return getCurrentFilteredRecords().filter(function (record) {
				return record.status === 'success' && Boolean(record.data);
			});
		}

		function normalizeExportValue(value) {
			if (value === undefined || value === null) {
				return '';
			}
			return String(value).trim();
		}

		function getNestedExportValue(source, path) {
			const parts = path.split('.');
			let current = source;
			for (const part of parts) {
				if (current === undefined || current === null || !Object.prototype.hasOwnProperty.call(current, part)) {
					return '';
				}
				current = current[part];
			}
			return normalizeExportValue(current);
		}

		function getProbeForTextExport(data, stackName) {
			const probe = data?.probe_results?.[stackName];
			return probe?.ok && probe.exit ? probe : null;
		}

		function getTextExportProbeCandidates(data) {
			if (activePrimaryFilter === 'only_ipv4') {
				return [getProbeForTextExport(data, 'ipv4'), getProbeForTextExport(data, 'ipv6')].filter(Boolean);
			}
			if (activePrimaryFilter === 'only_ipv6') {
				return [getProbeForTextExport(data, 'ipv6'), getProbeForTextExport(data, 'ipv4')].filter(Boolean);
			}
			return [getProbeForTextExport(data, 'ipv6'), getProbeForTextExport(data, 'ipv4')].filter(Boolean);
		}

		function getPreferredTextExportProbe(data) {
			const candidates = getTextExportProbeCandidates(data);
			if (activeCountryFilter !== 'all') {
				const countryMatchedProbe = candidates.find(function (probe) {
					return getExitCountryFilterKey(probe.exit) === activeCountryFilter;
				});
				if (countryMatchedProbe) {
					return countryMatchedProbe;
				}
			}
			return candidates[0] || null;
		}

		function getExportPingMs(data) {
			const raw = data?.responseTime;
			const num = typeof raw === 'number' ? raw : Number.parseFloat(String(raw || '').replace(/[^0-9.]/g, ''));
			return Number.isFinite(num) ? Math.round(num) : '';
		}

		function buildTextExportLine(data, detailLevel) {
			const proxyIP = normalizeExportValue(data?.proxyIP);
			const portRemote = normalizeExportValue(data?.portRemote);
			if (!proxyIP) {
				return '';
			}
			const level = detailLevel || 'full';
			if (level === 'ip') {
				return proxyIP;
			}
			const hostPort = portRemote ? (proxyIP + ':' + portRemote) : proxyIP;
			if (level === 'ip_port') {
				return hostPort;
			}
			// full details
			const exitData = getPreferredTextExportProbe(data)?.exit || {};
			const country = normalizeExportValue(exitData.country);
			const city = normalizeExportValue(exitData.city);
			const asn = normalizeExportValue(exitData.asn);
			const asOrganization = normalizeExportValue(exitData.asOrganization);
			const ping = getExportPingMs(data);
			const probes = Array.isArray(data?.tests_run) ? data.tests_run.join('+') : '';
			const passed = data?.probe_results && typeof data.probe_results === 'object'
				? Object.keys(data.probe_results).filter(function (k) { return data.probe_results[k] && data.probe_results[k].ok; }).join('+')
				: '';
			const description = [country, city, asn ? 'AS' + asn : '', asOrganization, ping !== '' ? (ping + 'ms') : '', passed ? ('ok:' + passed) : ''].filter(Boolean).join(' ');
			return hostPort + (description ? '#' + description : '');
		}

		function buildTextExport(records, detailLevel) {
			return records.map(function (record) {
				return buildTextExportLine(record.data, detailLevel);
			}).filter(Boolean).join('\\n');
		}

		function buildJsonExport(records, detailLevel) {
			const level = detailLevel || 'full';
			const rows = records.map(function (record) {
				const data = record.data || {};
				const proxyIP = normalizeExportValue(data.proxyIP);
				const portRemote = normalizeExportValue(data.portRemote);
				if (level === 'ip') {
					return { ip: proxyIP };
				}
				if (level === 'ip_port') {
					return { ip: proxyIP, port: portRemote ? Number(portRemote) || portRemote : '' };
				}
				const exitData = getPreferredTextExportProbe(data)?.exit || {};
				return {
					ip: proxyIP,
					port: portRemote ? Number(portRemote) || portRemote : '',
					ping_ms: getExportPingMs(data),
					country: normalizeExportValue(exitData.country),
					city: normalizeExportValue(exitData.city),
					asn: normalizeExportValue(exitData.asn),
					org: normalizeExportValue(exitData.asOrganization),
					passed_probes: record.passedProbes || [],
					supports_ipv4: !!data.supports_ipv4,
					supports_ipv6: !!data.supports_ipv6
				};
			}).filter(function (row) { return row.ip; });
			return JSON.stringify(rows, null, 2);
		}

		function askExportDetailLevel() {
			const answer = window.prompt(
				'Export detail level:\\n1 = IP only\\n2 = IP:Port\\n3 = Full details (includes ping)\\n\\nEnter 1, 2, or 3:',
				'3'
			);
			if (answer === null) return null; // cancelled
			const normalized = String(answer).trim();
			if (normalized === '1' || /^ip$/i.test(normalized)) return 'ip';
			if (normalized === '2' || /port/i.test(normalized)) return 'ip_port';
			return 'full';
		}

		function escapeCsvValue(value) {
			const text = normalizeExportValue(value);
			if (!/[",\\r\\n]/.test(text)) {
				return text;
			}
			return '"' + text.replace(/"/g, '""') + '"';
		}

		function buildCsvExport(records, detailLevel) {
			const level = detailLevel || 'full';
			let columns;
			if (level === 'ip') {
				columns = [{ header: 'IP', path: 'proxyIP' }];
			} else if (level === 'ip_port') {
				columns = [
					{ header: 'IP', path: 'proxyIP' },
					{ header: 'PORT', path: 'portRemote' }
				];
			} else {
				columns = EXPORT_CSV_COLUMNS;
			}
			const headerLine = columns.map(function (column) {
				return escapeCsvValue(column.header);
			}).join(',');
			const rows = records.map(function (record) {
				return columns.map(function (column) {
					return escapeCsvValue(getNestedExportValue(record.data, column.path));
				}).join(',');
			});
			return [headerLine].concat(rows).join('\\n');
		}

		function downloadTextFile(content, filename, mimeType) {
			const blob = new Blob([content], { type: mimeType + ';charset=utf-8' });
			const url = URL.createObjectURL(blob);

			link.download = filename;
			link.style.display = 'none';
			document.body.appendChild(link);
			link.click();

			window.setTimeout(function () {
				URL.revokeObjectURL(url);
			}, 1000);
		}

		async function writeTextToClipboard(text) {
			if (navigator.clipboard && window.isSecureContext) {
				await navigator.clipboard.writeText(text);
				return;
			}

			const textArea = document.createElement('textarea');
			textArea.value = text;
			textArea.setAttribute('readonly', '');
			textArea.style.position = 'fixed';
			textArea.style.top = '-1000px';
			textArea.style.left = '-1000px';
			document.body.appendChild(textArea);
			textArea.select();
			const didCopy = document.execCommand('copy');
			textArea.remove();
			if (!didCopy) {
				throw new Error('Clipboard copy command failed');
			}
		}

		function openFOFA() {
			const region = getSelectedFOFARegion();
			const port = proxyPortSelect ? proxyPortSelect.value : '';

			if (region === null) return;

			if (!region) {
				showToast('Please select a valid region', 'error');
				return;
			}

			let regionQuery = '';
			if (region === 'HK' || region === 'TW' || region === 'MO') {
				regionQuery = 'region="' + region + '"';
			} else {
				regionQuery = 'country="' + region + '"';
			}

			let portQuery = '';
			if (port === '443') {
				portQuery = 'port="443"';
			} else if (port === 'nonstandard') {
				portQuery = '(port!="80" && port!="8080" && port!="8880" && port!="2052" && port!="2082" && port!="2086" && port!="2095" && port!="443" && port!="2053" && port!="2083" && port!="2087" && port!="2096" && port!="8443")';
			} else {
				showToast('Please select a valid port', 'error');
				return;
			}

			const query = 'server=="cloudflare" && header="Forbidden" && asn!="13335" && asn!="209242" && ' + regionQuery + ' && ' + portQuery;
			const qbase64 = btoa(query);
			const url = 'https://fofa.info/result?qbase64=' + encodeURIComponent(qbase64);
			window.open(url, '_blank', 'noopener');
		}

		async function handleExport(format) {
			const records = getExportableRecords();
			if (!records.length) {
				showExportToast('No valid results to export in the current filter', 'error');
				return;
			}

			const detailLevel = askExportDetailLevel();
			if (!detailLevel) {
				showExportToast('Export cancelled');
				return;
			}

			try {
				if (format === 'csv') {
					downloadTextFile('\\ufeff' + buildCsvExport(records, detailLevel), getExportFileName('csv'), 'text/csv');
					showExportToast('CSV file download started (' + detailLevel + ')');
					return;
				}

				if (format === 'json') {
					const jsonContent = buildJsonExport(records, detailLevel);
					downloadTextFile(jsonContent, getExportFileName('json'), 'application/json');
					showExportToast('JSON file download started (' + detailLevel + ')');
					return;
				}

				const textContent = buildTextExport(records, detailLevel);
				if (!textContent) {
					showExportToast('No TXT content to export in the current filter', 'error');
					return;
				}

				if (format === 'clipboard') {
					await writeTextToClipboard(textContent);
					showExportToast('Copied to clipboard (' + detailLevel + ')');
					return;
				}

				if (format === 'txt') {
					downloadTextFile(textContent, getExportFileName('txt'), 'text/plain');
					showExportToast('TXT file download started (' + detailLevel + ')');
				}
			} catch (error) {
				console.error('Failed to export results', error);
				showExportToast(format === 'clipboard' ? 'Clipboard write failed — check your browser permissions' : 'Export failed — please try again later', 'error');
			}
		}

		function clampTimeoutMs(value, fallback, minMs, maxMs) {
			const n = Number.parseInt(String(value ?? ''), 10);
			if (!Number.isFinite(n)) return fallback;
			// 0 (or negative) is treated as 1ms so "instant fail" tests still hit the timer path
			const effective = n <= 0 ? 1 : n;
			return Math.min(maxMs, Math.max(minMs, effective));
		}

		function getSelectedTimeouts() {
			const tcpEl = document.getElementById('tcpTimeoutInput');
			const tlsEl = document.getElementById('tlsTimeoutInput');
			// Min 1ms so aggressive tests (e.g. tcp=1) actually time out instead of being raised to 500ms
			const tcpTimeoutMs = clampTimeoutMs(tcpEl && tcpEl.value, 5000, 1, 60000);
			const tlsTimeoutMs = clampTimeoutMs(tlsEl && tlsEl.value, 8000, 1, 60000);
			const httpTimeoutMs = Math.max(tcpTimeoutMs, tlsTimeoutMs, 1000);
			return { tcpTimeoutMs, tlsTimeoutMs, httpTimeoutMs };
		}

			const { wantIpv4, wantIpv6 } = getSelectedTestFlags();

			const probes = getSelectedProbeTargets();
			const params = [
				'testIpv4=' + (wantIpv4 ? '1' : '0'),
				'testIpv6=' + (wantIpv6 ? '1' : '0'),
				'testConnect=' + (wantConnect ? '1' : '0'),
				'testTls=' + (wantTls ? '1' : '0'),
				'testHttp=' + (wantHttp ? '1' : '0'),
				'tcpTimeoutMs=' + tcpTimeoutMs,
				'tlsTimeoutMs=' + tlsTimeoutMs,
				'timeoutMs=' + httpTimeoutMs,
				'probes=' + encodeURIComponent(JSON.stringify(probes))
			];
			return '&' + params.join('&');
		}

		async function checkIPBatch(jobs, run) {
			if (isRunStopped(run) || !jobs || !jobs.length) return;

			jobs.forEach(function (job) {
				job.itemObj = job.itemObj || addResultItem(job.target);
			});

			const proxyips = jobs.map(function (job) { return job.target; });
			const { wantIpv4, wantIpv6 } = getSelectedTestFlags();
			const { wantConnect, wantTls, wantHttp } = getSelectedSubstepFlags();
			const { tcpTimeoutMs, tlsTimeoutMs, httpTimeoutMs } = getSelectedTimeouts();
			const probes = getSelectedProbeTargets();
			const probeCount = Math.max(1, probes.length);
			// Client abort budget: enough for sequential worst-case per-IP TCP+TLS+HTTP across the chunk
			const perIpBudget = tcpTimeoutMs + tlsTimeoutMs + httpTimeoutMs + 2000;
			const timeoutMs = Math.min(180000, Math.max(30000, perIpBudget * jobs.length * probeCount));

			const chunkSeq = ++diagnosticsChunkSeq;
			const reqStartedAt = (window.performance && performance.now) ? performance.now() : Date.now();
			logDiagnostic({
				type: 'request',
				chunk: chunkSeq,
				chunkSize: jobs.length,
				candidates: proxyips,
				probeCount: probeCount,
				clientTimeoutMs: timeoutMs,
				tcpTimeoutMs: tcpTimeoutMs,
				tlsTimeoutMs: tlsTimeoutMs
			});

			try {
				const result = await fetchJsonWithTimeout('/check', {
					method: 'POST',
					headers: { 'Content-Type': 'application/json' },
					body: JSON.stringify({
						proxyips: proxyips,
						testIpv4: wantIpv4 ? 1 : 0,
						testIpv6: wantIpv6 ? 1 : 0,
						testConnect: wantConnect ? 1 : 0,
						testTls: wantTls ? 1 : 0,
						testHttp: wantHttp ? 1 : 0,
						tcpTimeoutMs: tcpTimeoutMs,
						tlsTimeoutMs: tlsTimeoutMs,
						timeoutMs: httpTimeoutMs,
						probes: probes
					})
				}, timeoutMs, run && run.controller ? run.controller.signal : undefined);

				if (isWorkerResourceFailure(result.response, result.payload)) {
					warnLowerIpsPerInvoke('HTTP ' + (result.response && result.response.status));
					logDiagnostic({
						type: 'chunk_error',
						chunk: chunkSeq,
						reason: 'worker_resource_failure',
						status: result.response && result.response.status,
						durationMs: Math.round(((window.performance && performance.now) ? performance.now() : Date.now()) - reqStartedAt)
					});
					jobs.forEach(function (job) {
						applyInvokeFailureToItem(job.target, job.itemObj, run, 'Worker invoke failed (CPU/wall-time). Lower “IPs per invoke”.');
					});
					updateProgress();
					updateResultFilters();
					return;
				}

				if (!result.response.ok) {
					const errMsg = (result.payload && (result.payload.error || result.payload.message)) || ('HTTP ' + result.response.status);
					if (isWorkerResourceFailure(result.response, result.payload) || /cpu|time.?limit|wall/i.test(String(errMsg))) {
						warnLowerIpsPerInvoke(errMsg);
					} else {
						if (typeof showToast === 'function') showToast('Batch check failed: ' + errMsg, 'error');
					}
					logDiagnostic({
						type: 'chunk_error',
						chunk: chunkSeq,
						reason: 'http_error',
						status: result.response.status,
						message: String(errMsg),
						durationMs: Math.round(((window.performance && performance.now) ? performance.now() : Date.now()) - reqStartedAt)
					});
					jobs.forEach(function (job) {
						applyInvokeFailureToItem(job.target, job.itemObj, run, errMsg);
					});
					updateProgress();
					updateResultFilters();
					return;
				}

				let list = result.payload;
				if (!Array.isArray(list)) {
					list = list ? [list] : [];
				}

				logDiagnostic({
					type: 'response',
					chunk: chunkSeq,
					status: result.response.status,
					durationMs: Math.round(((window.performance && performance.now) ? performance.now() : Date.now()) - reqStartedAt),
					resultCount: list.length
				});

				const byCandidate = new Map();
				list.forEach(function (row) {
					if (!row) return;
					const key = String(row.candidate || row.proxyIP || '').trim();
					if (key) byCandidate.set(key, row);
					// also index host:port variants
					if (row.proxyIP != null && row.portRemote != null) {
						byCandidate.set(String(row.proxyIP) + ':' + String(row.portRemote), row);
					}
				});

				jobs.forEach(function (job, posInChunk) {
					let data = byCandidate.get(job.target);
					if (!data) {
						// try match without brackets differences
						const alt = job.target.replace(/^\[(.*)\]/, '$1');
						data = byCandidate.get(alt);
					}
					if (!data) {
						data = {
							success: false,
							message: 'No result returned for this IP in the batch (invoke may have been truncated). Try lower “IPs per invoke”.',
							candidate: job.target
						};
					}
					logDiagnostic({
						type: 'result',
						chunk: chunkSeq,
						chunkSize: jobs.length,
						posInChunk: posInChunk,
						candidate: job.target,
						ok: !!data.success,
						missing: !byCandidate.has(job.target),
						failStage: data.stages && data.stages.fail_stage,
						errorCode: data.stages && data.stages.error_code,
						errorBrief: data.stages && data.stages.error_brief,
						tcpOk: data.stages && data.stages.tcp_ok,
						tlsOk: data.stages && data.stages.tls_ok,
						tcpMs: data.stages && data.stages.tcp_ms,
						tlsMs: data.stages && data.stages.tls_ms,
						responseTime: data.responseTime,
						colo: data.colo
					});
					applyCheckResultToItem(job.target, job.itemObj, data, run);
				});
			} catch (error) {
				if (isRunStopped(run)) {
					jobs.forEach(function (job) {
						const itemObj = job.itemObj;
						const resultRecord = itemObj && itemObj.record;
						if (resultRecord) resultRecord.status = 'stopped';
						if (!itemObj) return;
						itemObj.badge.className = 'status-badge status-error';
						itemObj.badge.innerText = 'Stopped';
						itemObj.info.innerHTML =
							'<span class="result-label">Candidate Target</span>' +
							buildCopyableTarget(job.target) +
							'<span class="result-detail">Check was stopped manually; no further request was sent for this target.</span>';
						itemObj.meta.innerHTML = buildMetaChip('Stopped', 'info');
						itemObj.exitList.innerHTML = '';
					});
					updateProgress();
					updateResultFilters();
					return;
				}

				const isTimeout = error && (error.name === 'AbortError' || /timeout/i.test(String(error.message || error)));
				if (isTimeout) {
					warnLowerIpsPerInvoke('Request timed out after ' + timeoutMs + 'ms');
				} else if (typeof showToast === 'function') {
					showToast('Batch check error: ' + String(error && error.message || error), 'error');
				}

				logDiagnostic({
					type: 'chunk_error',
					chunk: chunkSeq,
					reason: isTimeout ? 'client_timeout' : 'fetch_error',
					message: String(error && error.message || error || 'Request error'),
					durationMs: Math.round(((window.performance && performance.now) ? performance.now() : Date.now()) - reqStartedAt)
				});

				jobs.forEach(function (job) {
					applyInvokeFailureToItem(
						job.target,
						job.itemObj,
						run,
						isTimeout
							? 'Check request timed out — lower “IPs per invoke”'
							: String(error && error.message || error || 'Request error')
					);
				});
			}

			updateProgress();
			updateResultFilters();
		}

		function applyInvokeFailureToItem(target, itemObj, run, message) {
			itemObj = itemObj || addResultItem(target);
			const resultRecord = itemObj.record;
			completedCount++;
			updateResultRecordAsError(resultRecord, { success: false, message: message });
			itemObj.el.className = 'result-item error';
			updateResultFlag(itemObj, '');
			itemObj.badge.className = 'status-badge status-error';
			itemObj.badge.innerText = 'Failed';
			itemObj.info.innerHTML =
				'<span class="result-label">Candidate Target</span>' +
				buildCopyableTarget(target) +
				'<span class="result-detail">' + escapeHtml(message || 'Invoke failed') + '</span>';
			itemObj.meta.innerHTML =
				buildMetaChip('TCP ?', 'tcp', 'meta-chip-danger') +
				buildMetaChip('TLS ?', 'tls', 'meta-chip-danger') +
				buildMetaChip(message || 'Invoke failed', 'error', 'meta-chip-danger');
			itemObj.exitList.innerHTML = '';
		}

		function applyCheckResultToItem(target, itemObj, data, run) {
			itemObj = itemObj || addResultItem(target);
			const resultRecord = itemObj.record;
			if (!data) {
				data = { success: false, message: 'Empty result' };
			}
			completedCount++;

			if (data.success) {
				successCount++;
				itemObj.el.className = 'result-item success';
				const latency = formatLatency(data.responseTime);
				itemObj.badge.className = 'status-badge status-success';
				itemObj.badge.innerText = latency;
				setLatencyTooltip(itemObj.badge, data, latency);

				const exitIps = [];
				if (data.probe_results && typeof data.probe_results === 'object') {
					Object.keys(data.probe_results).forEach(function (probeName) {
						const probe = data.probe_results[probeName];
						if (probe && probe.ok && probe.exit && probe.exit.ip) {
							const family = (probe.exit.ipType || '').toLowerCase();
							const stack = family === 'ipv4' || family === 'ipv6' ? family : probeName;
							exitIps.push({ stack: stack, probe: probeName, ip: probe.exit.ip, exitData: probe.exit });
						}
					});
				}
				updateResultRecordAsSuccess(resultRecord, data, exitIps);

				const locations = joinUniqueValues(exitIps.map(function (entry) {
					return formatExitLocation(entry.exitData);
				}), 'Region unknown');
				const networks = joinUniqueValues(exitIps.map(function (entry) {
					return formatExitNetwork(entry.exitData);
				}), 'ASN / ISP unknown');
				const flagUrl = getFlagUrlFromExitIps(exitIps);

				updateResultFlag(itemObj, flagUrl);

				const siteSummaries = [];
				if (data.probe_results && typeof data.probe_results === 'object') {
					Object.keys(data.probe_results).forEach(function (probeName) {
						const probe = data.probe_results[probeName];
						if (!probe || probe.mode !== 'site') return;
						if (probe.ok) {
							siteSummaries.push(probeName + ' OK' + (probe.status_code ? ' (' + probe.status_code + ')' : ''));
						} else if (probe.blocked) {
							siteSummaries.push(probeName + ' BLOCKED');
						} else {
							siteSummaries.push(probeName + ' FAIL');
						}
					});
				}
				const detailText = siteSummaries.length
					? ('Proxy check finished. Site probes: ' + siteSummaries.join(', ') + '.')
					: 'Proxy verified successfully — you can view exit location, network info, and map distribution below.';
				itemObj.info.innerHTML =
					'<span class="result-label">Candidate Target</span>' +
					buildCopyableTarget(data.candidate || target) +
					'<span class="result-detail">' + escapeHtml(detailText) + '</span>';

				const metaParts = [
					buildMetaChip(locations, 'location'),
					buildMetaChip(networks, 'network'),
					buildMetaChip(exitIps.length + ' exit(s)', 'exits')
				];
				siteSummaries.forEach(function (summary) {
					const isBlocked = summary.indexOf('BLOCKED') !== -1 || summary.indexOf('FAIL') !== -1;
					metaParts.push(buildMetaChip(summary, 'site', isBlocked ? 'meta-chip-danger' : 'meta-chip-strong'));
				});
				itemObj.meta.innerHTML = metaParts.join('');

				renderExitList(itemObj.exitList, exitIps);
			} else {
				updateResultRecordAsError(resultRecord, data);
				itemObj.el.className = 'result-item error';
				updateResultFlag(itemObj, '');
				itemObj.badge.className = 'status-badge status-error';
				itemObj.badge.innerText = 'Unavailable';
				const stages = data.stages || {};
				let tcpOk = stages.tcp_ok;
				let tlsOk = stages.tls_ok;
				let tcpBrief = '';
				let tlsBrief = '';
				const probeNotes = [];
				if (data.probe_results && typeof data.probe_results === 'object') {
					Object.keys(data.probe_results).forEach(function (probeName) {
						const probe = data.probe_results[probeName];
						if (!probe) return;
						if (probe.tcp_ok) tcpOk = true;
						if (probe.tls_ok) tlsOk = true;
						if (!probe.tcp_ok && !tcpBrief) {
							tcpBrief = probe.error_brief || probe.error || 'TCP failed';
						}
						if (probe.tcp_ok && !probe.tls_ok && !tlsBrief) {
							tlsBrief = probe.error_brief || probe.error || 'TLS failed';
						}
						if (probe.blocked) {
							probeNotes.push(probeName + ' BLOCKED' + (probe.block_reason ? ': ' + probe.block_reason : ''));
						} else if (!probe.ok) {
							const why = probe.error_brief || probe.error || probe.fail_stage || 'failed';
							probeNotes.push(probeName + ': ' + why);
						}
					});
				}
				if (stages.error_brief) {
					if (stages.fail_stage === 'tcp' && !tcpBrief) tcpBrief = stages.error_brief;
					if (stages.fail_stage === 'tls' && !tlsBrief) tlsBrief = stages.error_brief;
				}
				const tcpLabel = tcpOk
					? ('TCP pass' + (stages.tcp_ms != null ? ' ' + Math.round(stages.tcp_ms) + 'ms' : ''))
					: ('TCP fail' + (tcpBrief ? ': ' + tcpBrief : ''));
				const tlsLabel = tlsOk
					? ('TLS pass' + (stages.tls_ms != null ? ' ' + Math.round(stages.tls_ms) + 'ms' : ''))
					: ('TLS fail' + (tlsBrief ? ': ' + tlsBrief : (tcpOk ? '' : ' (not reached)')));
				const reasonLine = stages.error_brief || data.message || probeNotes[0] || 'Probe did not succeed';
				const failDetail = 'Why: ' + reasonLine
					+ (probeNotes.length ? ' · ' + probeNotes.slice(0, 4).join(' · ') : '');
				itemObj.info.innerHTML =
					'<span class="result-label">Candidate Target</span>' +
					buildCopyableTarget(target) +
					'<span class="result-detail">' + escapeHtml(failDetail) + '</span>';
				const failMeta = [
					buildMetaChip(tcpLabel, 'tcp', tcpOk ? 'meta-chip-strong' : 'meta-chip-danger'),
					buildMetaChip(tlsLabel, 'tls', tlsOk ? 'meta-chip-strong' : 'meta-chip-danger')
				];
				if (stages.fail_stage) {
					failMeta.push(buildMetaChip('Stage: ' + stages.fail_stage, 'stage', 'meta-chip-danger'));
				}
				probeNotes.slice(0, 4).forEach(function (note) {
					failMeta.push(buildMetaChip(note, 'probe', 'meta-chip-danger'));
				});
				itemObj.meta.innerHTML = failMeta.join('');
				itemObj.exitList.innerHTML = '';
			}
		}

		async function showDetails(button, exitData) {
			const item = button.closest('.result-item');
			const container = item.querySelector('.map-container-wrapper');
			const isOpen = container.style.display === 'block';
			const nextSelectionKey = button.dataset.exitKey || getExitSelectionKey(exitData);
			const isSameSelection = isOpen && container.dataset.activeExitKey === nextSelectionKey;
			const currentToken = ++mapRenderToken;

			document.querySelectorAll('.map-container-wrapper').forEach(function (panel) {
				if (panel !== container) {
					panel.style.display = 'none';
					panel.dataset.activeExitKey = '';
				}
			});
			document.querySelectorAll('.exit-ip-btn.is-active').forEach(function (activeButton) {
				activeButton.classList.remove('is-active');
			});

			if (isSameSelection) {
				container.style.display = 'none';
				container.dataset.activeExitKey = '';
				return;
			}

			container.dataset.activeExitKey = nextSelectionKey;
			button.classList.add('is-active');
			initMap();
			container.appendChild(globalMap);
			container.style.display = 'block';

			setTimeout(async function () {
				if (currentToken !== mapRenderToken || container.style.display !== 'block') {
					return;
				}

				map.invalidateSize();

				const exitLocation = parseCoordinatePair(exitData?.loc);
				await loadCfLocations();
				if (currentToken !== mapRenderToken || container.style.display !== 'block') {
					return;
				}

				const cfLocation = getCfLocation(exitData?.colo);
				const cfCoordinates = cfLocation ? [cfLocation.lat, cfLocation.lon] : null;
				const hasExitLocation = isValidCoordinatePair(exitLocation);
				const hasCfLocation = isValidCoordinatePair(cfCoordinates);

				clearMapLayers();

				if (hasExitLocation) {
					const exitMarker = L.circleMarker(exitLocation, {
						radius: 8,
						weight: 2,
						color: '#34d399',
						fillColor: '#34d399',
						fillOpacity: 0.3
					}).addTo(map);
					exitMarker.bindPopup(createExitPopup(exitData));
					mapLayers.push(exitMarker);
				}

				if (hasCfLocation) {
					const cfMarker = L.circleMarker(cfCoordinates, {
						radius: 8,
						weight: 2,
						color: '#61dbff',
						fillColor: '#61dbff',
						fillOpacity: 0.28
					}).addTo(map);
					cfMarker.bindPopup(createCfPopup(cfLocation));
					mapLayers.push(cfMarker);
				}

				if (hasExitLocation && hasCfLocation) {
					const transitLine = L.polyline([exitLocation, cfCoordinates], {
						color: '#8be9ff',
						weight: 2,
						opacity: 0.85,
						dashArray: '8 6',
						renderer: mapSvgRenderer
					}).addTo(map);
					mapLayers.push(transitLine);
					applyArrowStyleToPolyline(transitLine);
					map.fitBounds([exitLocation, cfCoordinates], {
						padding: [36, 36],
						maxZoom: 6
					});
					return;
				}

				if (hasExitLocation) {
					map.setView(exitLocation, 6);
					return;
				}

				if (hasCfLocation) {
					map.setView(cfCoordinates, 5);
					return;
				}

				map.setView([20, 0], 2);
			}, 100);
		}

function derReadLength(buf, offset) {
	if (offset >= buf.length) throw new Error('DER: truncated length');
	const first = buf[offset];
	if (first < 0x80) return { length: first, offset: offset + 1 };
	const n = first & 0x7f;
	if (n === 0 || n > 4 || offset + 1 + n > buf.length) throw new Error('DER: bad length');
	let length = 0;
	for (let i = 0; i < n; i++) length = (length << 8) | buf[offset + 1 + i];

	if (offset >= buf.length) throw new Error('DER: truncated tag');

	const { length, offset: contentOffset } = derReadLength(buf, offset + 1);
	const end = contentOffset + length;
	if (end > buf.length) throw new Error('DER: truncated value');
	return { tag, length, contentOffset, end, value: buf.subarray(contentOffset, end) };
}

function derExpectTag(buf, offset, expectedTag) {
	const tlv = derReadTLV(buf, offset);
	if (tlv.tag !== expectedTag) throw new Error(`DER: expected tag 0x${expectedTag.toString(16)}, got 0x${tlv.tag.toString(16)}`);
	return tlv;
}

function derOidToString(bytes) {
	if (!bytes.length) return '';
	const parts = [Math.floor(bytes[0] / 40), bytes[0] % 40];
	let value = 0;
	for (let i = 1; i < bytes.length; i++) {
		value = (value << 7) | (bytes[i] & 0x7f);
		if ((bytes[i] & 0x80) === 0) {
			parts.push(value);
			value = 0;
		}
	}
	return parts.join('.');
}

function derDecodeIA5OrUTF8(tlv) {
	// 0x0c UTF8String, 0x13 PrintableString, 0x16 IA5String, 0x1e BMPString (skip), 0x0a ENUMERATED skip
	try {
		if (tlv.tag === 0x0c || tlv.tag === 0x13 || tlv.tag === 0x16 || tlv.tag === 0x14) {
			return new TextDecoder('utf-8', { fatal: false }).decode(tlv.value);
		}
		if (tlv.tag === 0x1e && tlv.value.length % 2 === 0) {
			// BMPString big-endian
			let s = '';
			for (let i = 0; i < tlv.value.length; i += 2) {
				s += String.fromCharCode((tlv.value[i] << 8) | tlv.value[i + 1]);
			}
			return s;
		}
	} catch { }
	return '';
}

function extractCertDnsNames(certDer) {
	const names = [];
	try {
		const cert = derExpectTag(certDer, 0, 0x30); // Certificate SEQUENCE
		const tbs = derExpectTag(certDer, cert.contentOffset, 0x30); // TBSCertificate
		let offset = tbs.contentOffset;
		// version [0] OPTIONAL
		if (certDer[offset] === 0xa0) {
			const ver = derReadTLV(certDer, offset);
			offset = ver.end;
		}
		// serialNumber
		offset = derReadTLV(certDer, offset).end;
		// signature algorithm
		offset = derReadTLV(certDer, offset).end;
		// issuer
		offset = derReadTLV(certDer, offset).end;
		// validity
		offset = derReadTLV(certDer, offset).end;
		// subject
		const subject = derReadTLV(certDer, offset);
		offset = subject.end;
		// subjectPublicKeyInfo
		offset = derReadTLV(certDer, offset).end;

		// Parse CN from subject RDN sequence
		if (subject.tag === 0x30) {
			let sOff = subject.contentOffset;
			while (sOff < subject.end) {
				const rdn = derReadTLV(certDer, sOff);
				sOff = rdn.end;
				if (rdn.tag !== 0x31) continue; // SET
				let setOff = rdn.contentOffset;
				while (setOff < rdn.end) {
					const attr = derReadTLV(certDer, setOff);
					setOff = attr.end;
					if (attr.tag !== 0x30) continue;
					let aOff = attr.contentOffset;
					const oidTlv = derReadTLV(certDer, aOff);
					aOff = oidTlv.end;
					if (oidTlv.tag !== 0x06) continue;
					const oid = derOidToString(oidTlv.value);
					if (oid === '2.5.4.3') { // commonName
						const valTlv = derReadTLV(certDer, aOff);
						const cn = derDecodeIA5OrUTF8(valTlv);
						if (cn) names.push(cn);
					}
				}
			}
		}

		// Optional: issuerUniqueID, subjectUniqueID, extensions [3]
		while (offset < tbs.end) {
			const tlv = derReadTLV(certDer, offset);
			offset = tlv.end;
			// extensions are context-specific [3] => 0xa3
			if (tlv.tag !== 0xa3) continue;
			const extSeq = derExpectTag(tlv.value, 0, 0x30);
			let eOff = extSeq.contentOffset;
			while (eOff < extSeq.end) {
				const ext = derReadTLV(tlv.value, eOff);
				eOff = ext.end;
				if (ext.tag !== 0x30) continue;
				let xOff = ext.contentOffset;
				const oidTlv = derReadTLV(tlv.value, xOff);
				xOff = oidTlv.end;
				if (oidTlv.tag !== 0x06) continue;
				const oid = derOidToString(oidTlv.value);
				// optional critical BOOLEAN
				if (tlv.value[xOff] === 0x01) {
					xOff = derReadTLV(tlv.value, xOff).end;
				}
				const valOctet = derReadTLV(tlv.value, xOff);
				if (valOctet.tag !== 0x04) continue;
				if (oid === '2.5.29.17') { // subjectAltName
					try {
						const sanSeq = derExpectTag(valOctet.value, 0, 0x30);
						let sanOff = sanSeq.contentOffset;
						while (sanOff < sanSeq.end) {
							const gen = derReadTLV(valOctet.value, sanOff);
							sanOff = gen.end;
							// dNSName is [2] context-specific primitive => 0x82
							if (gen.tag === 0x82) {
								const dns = new TextDecoder('utf-8', { fatal: false }).decode(gen.value);
								if (dns) names.push(dns);
							}
						}
					} catch { }
				}
			}
		}
	} catch (err) {
		// Parsing failure => no names
	}
	return names;
}

function hostnameMatchesCertName(hostname, certName) {
	const host = String(hostname || '').toLowerCase().replace(/\.$/, '');
	const name = String(certName || '').toLowerCase().replace(/\.$/, '');
	if (!host || !name) return false;
	if (host === name) return true;
	// Wildcard: *.example.com matches foo.example.com (one label only)
	if (name.startsWith('*.')) {
		const suffix = name.slice(1); // .example.com
		const hostSuffix = host.slice(host.indexOf('.'));
		if (hostSuffix === suffix && host.indexOf('.') > 0) {
			const left = host.slice(0, host.indexOf('.'));
			if (left && !left.includes('.')) return true;
		}
	}
	return false;
}

function verifyCertMatchesSNI(certDer, serverName) {
	if (!serverName) return true;
	const names = extractCertDnsNames(certDer);
	if (!names.length) return false;
	return names.some((n) => hostnameMatchesCertName(serverName, n));
}

/** Reject certs that are clearly not a real site cert (common on broken/malicious proxies). */
function certNamesLookBogus(names) {
	if (!names || !names.length) return false; // unparsable — do not fail solely on parse miss
	const badExact = new Set([
		'localhost',
		'localhost.localdomain',
		'local',
		'invalid',
		'example.com',
		'example.org',
		'example.net',
		'test',
		'linux',
		'ubuntu',
		'debian',
		'nginx',
		'apache',
		'localhost.com',
	]);
	const badSuffixes = ['.local', '.localhost', '.internal', '.lan', '.home', '.corp'];
	for (const raw of names) {
		const n = String(raw || '').toLowerCase().replace(/\.$/, '').trim();
		if (!n) continue;
		if (badExact.has(n)) return true;
		if (n.startsWith('*.') && badExact.has(n.slice(2))) return true;
		// pure IP in SAN is ok for some probes; 127.0.0.1 / ::1 are not
		if (n === '127.0.0.1' || n === '::1' || n === '0.0.0.0') return true;
		for (const suf of badSuffixes) {
			if (n === suf.slice(1) || n.endsWith(suf) || (n.startsWith('*.') && n.slice(2).endsWith(suf))) {
				return true;
			}
		}
	}
	return false;
}

	'not available in your country',
	'not available in your region',
	'not supported in your country',
	'not supported in your region',
	'country is not supported',
	'region is not supported',
	'not available in your country or region',
	'403. That',
];

// Extra flexible patterns (case-insensitive) for region / permission blocks
const DEFAULT_SITE_BLOCK_PATTERNS = [
	/your client does not have permission to get url/i,
	/that'?s an error/i,
	/gemini isn'?t (currently )?supported in your country/i,
	/gemini isn'?t available in (your|this) country/i,
	/gemini is not available in (your|this) country/i,
	/not (currently )?(supported|available) in your (country|region)/i,
	/(country|region) (is )?not supported/i,
	/access (is )?restricted.*region/i,
	/403\s*\.\s*that/i,
];

function buildProbeTarget(
    name,
    host,
    path = '/',
    mode = 'exit',
    alpn = null,
    tlsOnly = false
) {
    const safePath =
        path && String(path).startsWith('/')
            ? String(path)
            : '/' + String(path || '');

    const probeMode = mode === 'site' ? 'site' : 'exit';

    const accept = probeMode === 'site'
        ? 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8'
        : 'application/json, text/plain, */*';

    return {
        name: String(name || host || 'probe')
            .replace(/[^a-zA-Z0-9._-]/g, '_')
            .slice(0, 32) || 'probe',

        host: String(host || '').trim(),
        path: safePath,
        mode: probeMode,

        alpn: Array.isArray(alpn)
            ? alpn.filter(Boolean)
            : alpn
                ? [alpn]
                : null,

        tlsOnly: !!tlsOnly,

        request: enc.encode(
    `GET ${safePath} HTTP/1.1\r\n` +
    `Host: ${host}\r\n` +
    `Connection: keep-alive\r\n` +
    `Upgrade-Insecure-Requests: 1\r\n` +
    `User-Agent: Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/153.0.0.0 Safari/537.36\r\n` +
    `Accept: text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8\r\n` +
    `Sec-Ch-Ua: "Google Chrome";v="153", "Not_A Brand";v="8", "Chromium";v="153"\r\n` +
    `Sec-Ch-Ua-Mobile: ?0\r\n` +
    `Sec-Ch-Ua-Platform: "Windows"\r\n` +
    `Sec-Fetch-Site: none\r\n` +
    `Sec-Fetch-Mode: navigate\r\n` +
    `Sec-Fetch-User: ?1\r\n` +
    `Sec-Fetch-Dest: document\r\n` +
    `Accept-Encoding: gzip, deflate, br, zstd\r\n` +
    `Accept-Language: en-US,en;q=0.9\r\n` +
    `\r\n`
),
    };
}

const PROBE_TARGETS = DEFAULT_PROBE_TARGETS.map(([name, host, path]) => buildProbeTarget(name, host, path, 'exit'));

function parseProbeTargetsFromRequest(body, searchParams) {
    const raw = body.probes ?? searchParams.get('probes');
    if (!raw) return null;

    let list = raw;

    if (typeof raw === 'string') {
        try {
            list = JSON.parse(raw);
        } catch {
            return null;
        }
    }

    if (!Array.isArray(list) || !list.length) return null;

    const parsed = [];
    const seen = new Set();

    for (const item of list) {
        const host = String(item?.host || item?.hostname || '').trim();
        if (!host) continue;

        const name = String(item?.name || host).trim();
        const path = item?.path || '/';
        const mode = item?.mode === 'site' ? 'site' : 'exit';

        const probe = buildProbeTarget(
            name,
            host,
            path,
            mode,
            null,
            false
        );

        if (seen.has(probe.name)) {
            probe.name = probe.name + '_' + parsed.length;
        }

        seen.add(probe.name);
        parsed.push(probe);
    }

    return parsed.length ? parsed : null;
}



function detectSiteBlock(responseText, statusCode) {
	const text = String(responseText || '');
	const lower = text.toLowerCase();
	const isShortBody = text.length > 0 && text.length < 8000;

	// Strong Gemini / Google signals — always count
	const strongPatterns = [
		[/your client does not have permission to get url/i, 'client does not have permission to get URL'],
		[/403\s*\.\s*that'?s an error/i, "403. That's an error"],
		[/gemini isn'?t (currently )?supported in your country/i, "Gemini isn't supported in your country"],
		[/gemini isn'?t available in (your|this) country/i, "Gemini isn't available in this country"],
		[/gemini is not available in (your|this) country/i, 'Gemini is not available in this country'],
		[/not (currently )?(supported|available) in your (country|region)/i, 'not available in your country/region'],
		[/(country|region) (is )?not supported/i, 'country/region not supported'],
		[/you'?ve been blocked by network security/i, "You've been blocked by network security"],
		[/sorry, you have been blocked/i, 'Sorry, you have been blocked'],
	];
	for (const [pattern, reason] of strongPatterns) {
		if (pattern.test(text)) {
			return { blocked: true, reason };
		}
	}

	// Weaker phrases only on error status or short error-like bodies (avoid SPA false positives)
	if (statusCode >= 400 || isShortBody) {
		for (const phrase of DEFAULT_SITE_BLOCK_PHRASES) {
			if (text.includes(phrase) || lower.includes(String(phrase).toLowerCase())) {
				return { blocked: true, reason: phrase };
			}
		}
		for (const pattern of DEFAULT_SITE_BLOCK_PATTERNS) {
			const match = text.match(pattern);
			if (match) {
				return { blocked: true, reason: match[0] };
			}
		}
	}

	if (statusCode === 403 || statusCode === 429 || statusCode === 451) {
		return { blocked: true, reason: `HTTP ${statusCode}` };
	}
	return { blocked: false, reason: null };
}

function parsePositiveInt(value, fallback) {
	const parsed = Number.parseInt(`${value ?? ''}`, 10);
	return Number.isFinite(parsed) && parsed > 0 ? parsed : fallback;
}

function parseBooleanFlag(value) {
	if (value === undefined || value === null) return false;
	const text = `${value}`.trim().toLowerCase();
	return text === '1' || text === 'true' || text === 'yes' || text === 'on';
}

function concatBytes(chunks) {
	const merged = new Uint8Array(chunks.reduce((sum, { length }) => sum + length, 0));
	let offset = 0;
	for (const chunk of chunks) {
		merged.set(chunk, offset);
		offset += chunk.length;
	}
	return merged;
}

function withTimeout(promise, timeoutMs, label) {
	const ms = Math.max(1, Number(timeoutMs) || 1);
	let timer;
	return Promise.race([
		promise,
		new Promise((_, reject) => {
			timer = setTimeout(
				() => reject(new Error(`${label} timeout after ${ms}ms`)),
				ms
			);
		}),
	]).finally(() => clearTimeout(timer));
}

function indexOfBytes(haystack, needle, start = 0) {
	outer: for (let i = start; i <= haystack.length - needle.length; i++) {
		for (let j = 0; j < needle.length; j++) {
			if (haystack[i + j] !== needle[j]) {
				continue outer;
			}
		}
		return i;
	}
	return -1;
}

function pickExitIp(payload) {
	// Prefer the "ip" field in the probe JSON; some endpoints use "ipAddress" instead.
	const ip = payload?.ip ?? payload?.ipAddress;
	return typeof ip === 'string' && ip ? ip : null;
}

function checkJsonResponse(data, status = 200) {
	return new Response(JSON.stringify(data, null, 2), {
		status,
		headers: CHECK_RESPONSE_HEADERS,
	});
}

// Prefer the ipType the probe returns; fall back to inferring it from the exit IP string if absent.
function getExitFamily(result) {
	if (!result?.ok) return null;
	const ipType = `${result?.exit?.ipType ?? ''}`.toLowerCase();
	if (ipType === 'ipv4' || ipType === 'ipv6') return ipType;
	const exitIp = pickExitIp(result.exit) ?? '';
	// IPv4: must match x.x.x.x with each octet 0-255.
	if (/^\d{1,3}(?:\.\d{1,3}){3}$/.test(exitIp) && exitIp.split('.').every((part) => Number(part) <= 255)) return 'ipv4';
	// IPv6: current logic treats anything containing ":" as IPv6 (not a full RFC-strict check).
	if (exitIp.includes(':')) return 'ipv6';
	return null;
}

async function handleCheckProxyRequest(req) {
	if (req.method === 'OPTIONS') {
		return new Response(null, {
			status: 204,
			headers: CHECK_RESPONSE_HEADERS,
		});
	}

	if (req.method !== 'GET' && req.method !== 'POST') {
		return checkJsonResponse(
			{ success: false, error: 'method not allowed', usage: USAGE_EXAMPLES },
			405
		);
	}

	try {
		const url = new URL(req.url);
		const pathname = url.pathname.replace(/\/+$/, '') || '/';
		if (pathname !== '/check') {
			return checkJsonResponse(
				{ success: false, error: 'not found', usage: USAGE_EXAMPLES },
				404
			);
		}

		const { searchParams } = url;
		const bodyText = req.method === 'POST' ? await req.text() : '';
		const body = bodyText ? JSON.parse(bodyText) : {};

		const rawCandidates = body.proxyips ?? body.proxyip ?? searchParams.get('proxyip');

		const rawList = Array.isArray(rawCandidates) ? rawCandidates : `${rawCandidates ?? ''}`.split(/[\s,]+/);
		const candidates = [...new Set(rawList.map((item) => `${item ?? ''}`.trim()).filter(Boolean))];
		const timeoutMs = parsePositiveInt(body.timeoutMs ?? searchParams.get('timeoutMs'), DEFAULT_TIMEOUT_MS);
		// Per-IP stage timeouts (not summed across batch). Min 1ms so UI values like 1 are honored.
		// 0 / invalid → fall back to timeoutMs (or 1 if that is also invalid after clamp).
		const parseStageTimeout = (raw, fallback) => {
			const n = Number.parseInt(`${raw ?? ''}`, 10);
			if (!Number.isFinite(n)) return Math.min(60000, Math.max(1, fallback));
			if (n <= 0) return 1; // explicit zero → 1ms fail-fast for testing
			return Math.min(60000, Math.max(1, n));
		};
		const tcpTimeoutMs = parseStageTimeout(body.tcpTimeoutMs ?? searchParams.get('tcpTimeoutMs'), timeoutMs);
		const tlsTimeoutMs = parseStageTimeout(body.tlsTimeoutMs ?? searchParams.get('tlsTimeoutMs'), timeoutMs);
		const httpTimeoutMs = parseStageTimeout(body.httpTimeoutMs ?? searchParams.get('httpTimeoutMs'), timeoutMs);
		const readLimit = parsePositiveInt(body.readLimit ?? searchParams.get('readLimit'), DEFAULT_READ_LIMIT);

		if (!candidates.length) {
			return checkJsonResponse(
				{ success: false, error: 'missing proxyip', usage: USAGE_EXAMPLES },
				400
			);
		}

		const MAX_CANDIDATES_PER_INVOKE = 50;
		if (candidates.length > MAX_CANDIDATES_PER_INVOKE) {
			return checkJsonResponse(
				{
					success: false,
					error: `too many proxyips in one invoke (max ${MAX_CANDIDATES_PER_INVOKE})`,
					hint: 'Lower “IPs per invoke” in the UI',
				},
				400
			);
		}

		// Each candidate can open up to ~2 sockets at once (ipv4 + ipv6 exit
		// probes run concurrently; site probes only start after those finish).
		// Cloudflare Workers cap simultaneous open connections per invocation
		// at 6, so keep candidateConcurrency low enough that
		// candidateConcurrency * 2 stays at/under that cap by default.
		const DEFAULT_CANDIDATE_CONCURRENCY = 3;
		const MAX_CANDIDATE_CONCURRENCY = 6;
		const candidateConcurrency = Math.max(
			1,
			Math.min(
				MAX_CANDIDATE_CONCURRENCY,
				parsePositiveInt(body.probeConcurrency ?? searchParams.get('probeConcurrency'), DEFAULT_CANDIDATE_CONCURRENCY)
			)
		);

		const customProbes = parseProbeTargetsFromRequest(body, searchParams);

const baseProbes = customProbes
    ? customProbes.map((target) => {
        if (target.mode !== 'site') return target;

        return {
            ...target,
            alpn: ['h2', 'http/1.1'],
            tlsOnly: true,
            browserCompatible: true
        };
    })
    : PROBE_TARGETS;

		// Optional stack filter still applies when using default ipv4/ipv6 named probes.
		const testIpv4Raw = body.testIpv4 ?? searchParams.get('testIpv4');
		const testIpv6Raw = body.testIpv6 ?? searchParams.get('testIpv6');
		const hasExplicitTestSelection = testIpv4Raw !== undefined && testIpv4Raw !== null
			|| testIpv6Raw !== undefined && testIpv6Raw !== null;
		const wantIpv4Test = parseBooleanFlag(testIpv4Raw);
		const wantIpv6Test = parseBooleanFlag(testIpv6Raw);

		let selectedProbeTargets = baseProbes;
		if (hasExplicitTestSelection && !customProbes) {
			selectedProbeTargets = baseProbes.filter((target) =>
				(target.name === 'ipv4' && wantIpv4Test) || (target.name === 'ipv6' && wantIpv6Test)
			);
		} else if (hasExplicitTestSelection && customProbes) {
			// When custom probes are supplied, stack checkboxes filter by exit family later;
			// still require at least one stack selected.
			if (!wantIpv4Test && !wantIpv6Test) {
				selectedProbeTargets = [];
			}
		}
         

		if (!selectedProbeTargets.length) {
			return checkJsonResponse(
				{ success: false, error: 'no test selected (testIpv4/testIpv6 or probes)', usage: USAGE_EXAMPLES },
				400
			);
		}

		const testConnectRaw = body.testConnect ?? searchParams.get('testConnect');
		const testTlsRaw = body.testTls ?? searchParams.get('testTls');
		const testHttpRaw = body.testHttp ?? searchParams.get('testHttp');
		const hasSubstepSelection = testConnectRaw !== undefined && testConnectRaw !== null
			|| testTlsRaw !== undefined && testTlsRaw !== null
			|| testHttpRaw !== undefined && testHttpRaw !== null;
		const wantConnect = hasSubstepSelection ? parseBooleanFlag(testConnectRaw) : true;
		const wantTls = hasSubstepSelection ? parseBooleanFlag(testTlsRaw) : true;
		const wantHttp = hasSubstepSelection ? parseBooleanFlag(testHttpRaw) : true;

		if (!wantConnect && !wantTls && !wantHttp) {
			return checkJsonResponse(
				{ success: false, error: 'no sub-step selected (testConnect/testTls/testHttp)', usage: USAGE_EXAMPLES },
				400
			);
		}

		const results = await mapWithConcurrency(
			candidates,
			candidateConcurrency,
			async (rawCandidate) => {
				const defaultPort = 443;
				let candidate;
				if (rawCandidate.startsWith('[')) {
					const match = rawCandidate.match(/^\[([^\]]+)\](?::(\d+))?$/);
					if (!match) throw new Error(`invalid IPv6 candidate: ${rawCandidate}`);
					candidate = { raw: rawCandidate, hostname: match[1], port: Number(match[2]) || defaultPort };
				} else {
					const [, hostname = rawCandidate, port] = rawCandidate.match(/^([^:]+):(\d+)$/) ?? [];
					candidate = { raw: rawCandidate, hostname, port: Number(port) || defaultPort };
				}

				const runProbe = async (target) => {
						let socket = null;
						let tlsClient = null;

						let connectMs = null;
let tlsMs = null;
let httpMs = null;
let statusCode = null;

let tlsSucceeded = false;

						const classifyStageError = (raw) => {
							const msg = String(raw || '').trim();
							const lower = msg.toLowerCase();
							if (!msg) return { code: 'unknown', brief: 'unknown error' };
							if (/certificate looks dummy|localhost|127\.0\.0\.1|::1/i.test(msg)) {
								return { code: 'tls_dummy_cert', brief: 'TLS dummy cert (e.g. localhost)' };
							}
							if (/certificate does not match|could not parse dns names/i.test(msg)) {
								return { code: 'tls_cert_name', brief: 'TLS cert name invalid' };
							}
							if (/close.?notify/i.test(lower)) {
    return {
        code: 'tls_close_notify',
        brief: 'TLS peer sent close_notify during handshake'
    };
}
							if (/tls alert|handshake|certificate|alert:/i.test(msg) && /tls/i.test(msg)) {
								return { code: 'tls_handshake', brief: 'TLS handshake failed' };
							}
							if (/tls handshake/.test(lower) && /timeout/.test(lower)) {
								return { code: 'tls_timeout', brief: 'TLS handshake timeout' };
							}
							if (/tcp connect/.test(lower) && /timeout/.test(lower)) {
								return { code: 'tcp_timeout', brief: 'TCP timeout (no response)' };
							}
							if (/timeout/.test(lower) && connectMs == null) {
								return { code: 'tcp_timeout', brief: 'TCP timeout (no response)' };
							}
							if (/timeout/.test(lower) && tlsMs == null) {
								return { code: 'tls_timeout', brief: 'TLS timeout' };
							}
							if (/refused|econnrefused/.test(lower)) {
								return { code: 'tcp_refused', brief: 'TCP connection refused' };
							}
							if (/reset|econnreset/.test(lower)) {
								return { code: 'tcp_reset', brief: 'TCP connection reset' };
							}
							if (/unreachable|network|no route|enonet|ehostunreach/.test(lower)) {
								return { code: 'tcp_unreachable', brief: 'Host unreachable (dead path)' };
							}
							if (/closed|connection closed/.test(lower) && connectMs == null) {
								return { code: 'tcp_closed', brief: 'TCP closed before open' };
							}
							if (/closed|connection closed/.test(lower)) {
								return { code: 'conn_closed', brief: 'Connection closed during handshake' };
							}
							if (/site blocked/i.test(msg)) {
								return { code: 'site_blocked', brief: msg.replace(/^site blocked:\s*/i, 'Blocked: ').slice(0, 80) };
							}
							if (/empty response/.test(lower)) {
								return { code: 'http_empty', brief: 'HTTP empty response' };
							}
							if (/unexpected status|site unexpected/.test(lower)) {
								return { code: 'http_status', brief: msg.slice(0, 80) };
							}
							if (/invalid json|missing exit ip/.test(lower)) {
								return { code: 'exit_parse', brief: 'Exit probe response invalid' };
							}
							if (/skipped: exit/i.test(msg)) {
								return { code: 'skipped', brief: 'Skipped (exit probe failed)' };
							}
							return { code: 'other', brief: msg.slice(0, 100) };
						};

						const buildResult = (
							ok,
							resultStatusCode,
							error,
							{ exit = null, blocked = false, block_reason = null, body_preview = null, mode = target.mode } = {}
						) => {
							const tcpOk = connectMs != null;
const tlsOk = tlsSucceeded;
const httpOk = httpMs != null && ok;
							let failStage = null;
							if (!ok) {
								if (!tcpOk) failStage = 'tcp';
								else if (wantTls && !tlsOk) failStage = 'tls';
								else if (wantHttp) failStage = mode === 'site' ? 'site' : 'http';
								else failStage = 'other';
							}
							const classified = error ? classifyStageError(error) : null;
							return {
								candidate: candidate.raw,
								name: target.name,
								mode,
								connect_ms: connectMs,
								tls_ms: tlsMs,
								http_ms: httpMs,
								status_code: resultStatusCode,
								ok,
								error,
								error_brief: classified ? classified.brief : null,
								error_code: classified ? classified.code : null,
								fail_stage: failStage,
								tcp_ok: tcpOk,
								tls_ok: tlsOk,
								http_ok: !!httpOk,
								exit,
								blocked,
								block_reason,
								body_preview,
							};
						};

						try {
							if (!wantConnect) {
								return buildResult(false, null, 'tcp connect step disabled');
							}
							let startedAt = Date.now();
							const hostAddr = candidate.hostname.includes(':') ? `[${candidate.hostname}]` : candidate.hostname;
							socket = connect({ hostname: hostAddr, port: candidate.port });
							// Per-IP TCP timeout — fail this IP early; close socket so the dial does not keep running
							try {
								await withTimeout(socket.opened, tcpTimeoutMs, 'tcp connect');
							} catch (tcpErr) {
								try { socket.close(); } catch { }
								socket = null;
								throw tcpErr;
							}
							connectMs = Date.now() - startedAt;

							if (!wantTls) {
								return buildResult(true, null, null, { exit: null });
							}

							startedAt = Date.now();
							// SNI = probe host; cert must match this host (SAN or CN). TLS timeout is per-IP / per-probe.
							tlsClient = new TlsClient(socket, { serverName: target.host, timeout: tlsTimeoutMs, alpn: target.alpn });
							await withTimeout(
    tlsClient.handshake(),
    tlsTimeoutMs,
    'tls handshake'
);

tlsMs = Date.now() - startedAt;
tlsSucceeded = true;

if (target.browserCompatible) {
    const negotiated = tlsClient.negotiatedAlpn || null;

    if (negotiated !== 'h2' && negotiated !== 'http/1.1') {
        return buildResult(
            false,
            null,
            `browser TLS: unacceptable ALPN negotiation: ${negotiated || 'none'}`,
            { mode: target.mode }
        );
    }

    return buildResult(
        true,
        null,
        null,
        { mode: target.mode }
    );
}

if (target.tlsOnly) {
    return buildResult(true, null, null, {
        mode: target.mode
    });
}

if (!wantHttp) {
    return buildResult(true, null, null, { exit: null });
}

							startedAt = Date.now();
							await withTimeout(tlsClient.write(target.request), httpTimeoutMs, 'http write');
							const chunks = [];
							for (let read = 0; read < readLimit;) {
								const chunk = (await withTimeout(tlsClient.read(), httpTimeoutMs, 'http read'))?.subarray(0, readLimit - read);
								if (!chunk?.length) {
									break;
								}
								chunks.push(chunk);
								read += chunk.length;
							}
							const rawResponse = concatBytes(chunks);
							httpMs = Date.now() - startedAt;

							if (!rawResponse.length) return buildResult(false, null, 'empty response');

							const splitIndex = indexOfBytes(rawResponse, HEADER_BODY_SEPARATOR);
							const [headerBytes, bodyBytes] = splitIndex < 0
								? [rawResponse, new Uint8Array(0)]
								: [rawResponse.subarray(0, splitIndex), rawResponse.subarray(splitIndex + HEADER_BODY_SEPARATOR.length)];
							const headerText = dec.decode(headerBytes);
							statusCode = Number(headerText.match(HTTP_STATUS_RE)?.[1] ?? 0) || null;

							let responseBodyBytes;
							if (splitIndex < 0) {
								responseBodyBytes = rawResponse;
							} else if (CHUNKED_TRANSFER_RE.test(`\r\n${headerText}\r\n`)) {
								const decodedChunks = [];
								for (let offset = 0; offset < bodyBytes.length;) {
									const lineEnd = indexOfBytes(bodyBytes, CRLF, offset);
									// Truncated read (large sites like Gemini) — use what we have
									if (lineEnd < 0) {
										if (decodedChunks.length) break;
										responseBodyBytes = bodyBytes;
										break;
									}
									const sizeHex = dec.decode(bodyBytes.slice(offset, lineEnd)).split(';', 1)[0].trim();
									const size = Number.parseInt(sizeHex, 16);
									if (!Number.isFinite(size)) {
										if (decodedChunks.length) break;
										responseBodyBytes = bodyBytes;
										break;
									}
									const bodyStart = lineEnd + CRLF.length;
									const bodyEnd = bodyStart + size;
									if (!size) break;
									// Partial last chunk after readLimit — keep complete chunks only
									if (bodyEnd > bodyBytes.length) {
										decodedChunks.push(bodyBytes.slice(bodyStart));
										break;
									}
									decodedChunks.push(bodyBytes.slice(bodyStart, bodyEnd));
									offset = bodyEnd + CRLF.length;
								}
								if (!responseBodyBytes) responseBodyBytes = decodedChunks.length ? concatBytes(decodedChunks) : bodyBytes;
							} else {
								responseBodyBytes = bodyBytes;
							}
							const responseText = dec.decode(responseBodyBytes);
							const bodyPreview = responseText ? responseText.slice(0, 180).replace(/\s+/g, ' ').trim() : '';

							// Site-check mode: real website reachability + block-page detection
							if (target.mode === 'site') {
								const blockInfo = detectSiteBlock(responseText, statusCode);
								if (blockInfo.blocked) {
									return buildResult(false, statusCode, `site blocked: ${blockInfo.reason}`, {
										blocked: true,
										block_reason: blockInfo.reason,
										body_preview: bodyPreview,
										mode: 'site',
									});
								}
								// Only HTTP 200 is accepted (301/302/403/etc. fail)
								if (statusCode !== 200) {
									return buildResult(false, statusCode, `site unexpected status: ${statusCode ?? 'unknown'} (only 200 accepted)`, {
										body_preview: bodyPreview,
										mode: 'site',
									});
								}
								return buildResult(true, statusCode, null, {
									body_preview: bodyPreview || (statusCode ? ('HTTP ' + statusCode) : ''),
									mode: 'site',
								});
							}

							// Exit-JSON mode (default ProxyIP probe)
							if (statusCode !== 200) {
								return buildResult(false, statusCode, `unexpected status: ${statusCode ?? 'unknown'}${bodyPreview ? ' body: ' + bodyPreview : ''}`);
							}

							let payload;
							try {
								payload = JSON.parse(responseText);
							} catch (error) {
								return buildResult(false, statusCode, `invalid json response: ${String(error?.message || error)}`);
							}

							if (!pickExitIp(payload)) return buildResult(false, statusCode, 'probe json missing exit ip');
							return buildResult(true, statusCode, null, { exit: payload, mode: 'exit' });
						} catch (error) {
							return buildResult(false, statusCode, String(error?.message || error));
						} finally {
							try { tlsClient?.close(); } catch { }
							try { if (!tlsClient) socket?.close(); } catch { }
						}
				};

				const exitTargets = selectedProbeTargets.filter((t) => t.mode !== 'site');
				const siteTargets = selectedProbeTargets.filter((t) => t.mode === 'site');

				let probeResults = [];
				if (exitTargets.length) {
					const exitResults = await Promise.all(exitTargets.map(runProbe));
					probeResults = probeResults.concat(exitResults);
					const anyExitOk = exitResults.some((r) => r && r.ok);
					if (!anyExitOk) {
						// Exit/normal probes failed — skip site probes entirely
						for (const target of siteTargets) {
							probeResults.push({
								candidate: candidate.raw,
								name: target.name,
								mode: 'site',
								connect_ms: null,
								tls_ms: null,
								http_ms: null,
								status_code: null,
								ok: false,
								error: 'skipped: exit/normal probe failed',
								error_brief: 'Skipped (exit probe failed)',
								error_code: 'skipped',
								fail_stage: 'skipped',
								tcp_ok: false,
								tls_ok: false,
								http_ok: false,
								exit: null,
								blocked: false,
								block_reason: null,
								body_preview: null,
								skipped: true,
							});
						}
					} else if (siteTargets.length) {
						const siteResults = await Promise.all(siteTargets.map(runProbe));
						probeResults = probeResults.concat(siteResults);
					}
				} else if (siteTargets.length) {
					// Only site probes selected
					probeResults = await Promise.all(siteTargets.map(runProbe));
				}

				// probeResults order matches selectedProbeTargets: entries are looked up by probe name below
				const probeResultsByName = {};
				probeResults.forEach((result) => {
					if (!result) return;
					const key = result.name || 'probe';
					probeResultsByName[key] = result;
				});
				// Ensure every selected target has an entry
				selectedProbeTargets.forEach((target) => {
					if (!probeResultsByName[target.name]) {
						probeResultsByName[target.name] = {
							name: target.name,
							mode: target.mode || 'exit',
							ok: false,
							error: 'no result',
						};
					}
				});
				const ipv4 = probeResultsByName.ipv4 ?? null;
				const ipv6 = probeResultsByName.ipv6 ?? null;
				const hasIPv4 = getExitFamily(ipv4) === 'ipv4';
				const hasIPv6 = getExitFamily(ipv6) === 'ipv6';
				const displayedProbeResults = {};
				// Always include every probe result (exit + site) so the UI can filter and show site blocks
				for (const [name, result] of Object.entries(probeResultsByName)) {
					if (result) displayedProbeResults[name] = result;
				}
				const inferredStack = hasIPv4 && hasIPv6 ? 'dual_stack' : hasIPv4 ? 'ipv4_only' : hasIPv6 ? 'ipv6_only' : 'unknown';
				const responseTime = probeResults.length
					? Math.ceil(probeResults.reduce((s, r) => s + (Number.isFinite(r?.connect_ms) ? r.connect_ms : 0), 0) / probeResults.length)
					: 0;
				const siteResults = probeResults.filter((r) => r?.mode === 'site');
const siteOk = siteResults.filter((r) => r.ok).length;
const siteBlocked = siteResults.filter((r) => r.blocked).length;
const exitOk = probeResults.some((r) => r?.mode !== 'site' && r.ok);

let anyOk;

if (customProbes) {
    const customSiteResults = probeResults.filter(
        (result) => result?.mode === 'site'
    );

    anyOk =
        customSiteResults.length > 0 &&
        customSiteResults.every(
            (result) =>
                result.ok === true &&
                result.tcp_ok === true &&
                result.tls_ok === true
        );
} else {
    // Existing default behavior remains unchanged.
    anyOk = probeResults.some((result) => result.ok);
}

					// Aggregate TCP/TLS stage across probes (best effort / any success wins)
					const tcpAnyOk = probeResults.some((r) => r && r.tcp_ok);
					const tlsAnyOk = probeResults.some((r) => r && r.tls_ok);
					const firstFail = probeResults.find((r) => r && !r.ok) || null;
					const stageSummary = {
						tcp_ok: tcpAnyOk,
						tls_ok: tlsAnyOk,
						tcp_ms: probeResults.map((r) => r?.connect_ms).find((v) => v != null) ?? null,
						tls_ms: probeResults.map((r) => r?.tls_ms).find((v) => v != null) ?? null,
						fail_stage: firstFail?.fail_stage || null,
						error_brief: firstFail?.error_brief || firstFail?.error || null,
						error_code: firstFail?.error_code || null,
					};

				return {
					candidate: rawCandidate,
					success: anyOk,
					proxyIP: candidate.hostname,
					portRemote: candidate.port,
					inferred_stack: inferredStack,
					supports_ipv4: hasIPv4,
					supports_ipv6: hasIPv6,
					dual_stack: inferredStack === 'dual_stack',
					exit_ok: exitOk,
					site_ok_count: siteOk,
					site_blocked_count: siteBlocked,
					stages: stageSummary,
					responseTime,
					colo: req.cf?.colo || 'CF',
					timeStamp: new Date().toISOString(),
					probe_results: displayedProbeResults,
					tests_run: selectedProbeTargets.map((target) => target.name),
				};
			}
		);
		return checkJsonResponse(results.length === 1 ? results[0] : results);
	} catch (error) {
		const msg = String(error?.message || error);
		return checkJsonResponse({
			ok: false,
			success: false,
			error: msg,
			hint: 'If this is a CPU/time limit, lower “IPs per invoke” in the UI',
		}, 500);
	}
}
